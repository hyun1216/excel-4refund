Function GetDictionaryFromCell(ws As Worksheet, cellAddress As String) As Object
'셀에 있는 데이터를 딕셔너리에 반영
    Dim values            As String
    Dim dict                As Object
    Dim splitValues     As Variant

    Set dict = CreateObject("Scripting.Dictionary")

    values = Replace(ws.Range(cellAddress).value, " ", "")

    For Each splitValues In Split(values, ",")
        If Not dict.Exists(splitValues) Then
            dict.Add splitValues, splitValues
        End If
    Next splitValues

    Set GetDictionaryFromCell = dict
    
End Function
Function GetTextAfterLastUnderscore(targetText As Variant) As String
'텍스트에서 마지막 '_' 뒷 문자 추출(파일명일 경우 확장자명 삭제 포함)
    Dim pos As Integer
    Dim extPos As Integer
    Dim baseName As String
    
    extPos = InStrRev(targetText, ".")
    If extPos > 0 Then
        baseName = Left(targetText, extPos - 1)   ' 확장자 제거
    Else
        baseName = targetText                              ' 확장자가 없으면 그대로 사용
    End If
    
    pos = InStrRev(baseName, "_")                       '마지막 언더스코어 위치 찾기
    If pos > 0 Then
        GetTextAfterLastUnderscore = Mid(baseName, pos + 1) ' 언더스코어 뒤 문자 추출
    Else
        GetTextAfterLastUnderscore = ""                ' 언더스코어가 없으면 빈 문자열 반환
    End If
End Function
Function FindNth(ByVal str As String, ByVal char As String, ByVal n As Long) As Long
' 문자열에서 n번째 있는 특정 문자 위치 확인
    Dim i               As Long
    Dim count       As Long
    For i = 1 To Len(str)
        If Mid(str, i, 1) = char Then
            count = count + 1
            If count = n Then FindNth = i: Exit Function
        End If
    Next i
    FindNth = 0
End Function
Function FindRange(targetSheet As Worksheet, searchRng As Range, startKeyword As Variant, endKeyword As Variant, _
                                    ByRef startCell As Range, ByRef endCell As Range) As String
' 타겟범위에서 시작키워드, 종료키워드가 있는 셀 주소 반환
    
    Dim cell As Range
    Dim result As String

    Set startCell = Nothing
    Set endCell = Nothing

    For Each cell In searchRng
        If cell.value = startKeyword And startCell Is Nothing Then
            Set startCell = cell
        End If
        If cell.value = endKeyword And endCell Is Nothing Then
            Set endCell = cell
        End If
    Next cell
    '키워드가 아닌 셀주소를 직접 입력했을 경우
    If endCell Is Nothing Then
        Set endCell = targetSheet.Range(endKeyword)
    End If

    FindRange = result
    
End Function
Sub editTo1RowForm(targetWs As Worksheet, startDataCell As Range, endDataCell As Range, endDataRowOfWs As Long, rowCountPerPerson)
' 시작셀, 종료셀 위치 계산하여 개인별 데이터 한 행으로 수정

    Dim colCountOfData             As Integer
    Dim c                                      As Integer
    Dim r                                      As Integer
    Dim sourceCell                       As Range
    Dim rowFromStartData          As Long

    With targetWs
        colCountOfData = endDataCell.Column - startDataCell.Column + 1
                
        For r = startDataCell.Row + 1 To endDataRowOfWs
            For c = startDataCell.Column To endDataCell.Column
                Set sourceCell = .Cells(r, c)
                rowFromStartData = r - startDataCell.Row
                
                If rowFromStartData Mod rowCountPerPerson <> 0 Then
                    'sourceCell.Offset(-1 * (r Mod rowCountPerPerson), colCountOfData * (r Mod rowCountPerPerson)).Value = sourceCell.Value
                    sourceCell.Offset(-1 * (rowFromStartData Mod rowCountPerPerson), colCountOfData * (rowFromStartData Mod rowCountPerPerson)).value = sourceCell.value
                    sourceCell.ClearContents
                End If
            Next c
            
        Next r
    End With

End Sub

Sub DeleteFilesInFolder(folderPath As String)
' 경로에 있는 모든 파일 삭제
    Dim fileName As String
    
    fileName = Dir(folderPath & "\*.*")
    
    On Error Resume Next
    Do While fileName <> ""
        Kill folderPath & "\" & fileName
        fileName = Dir
    Loop
    On Error GoTo 0
End Sub
Function GenerateUniqueFileName(filePath As String, baseName As String) As String
' 파일명 중복일 경우 인덱스 추가한 파일명 생성
    Dim fileName As String
    Dim fileIndex As Integer

    fileName = filePath & "\" & baseName
    fileIndex = 1

    Do While Dir(fileName) <> ""
        fileName = filePath & "\" & Replace(baseName, ".xls", "") & " (" & fileIndex & ").xls"
        fileIndex = fileIndex + 1
    Loop

    GenerateUniqueFileName = fileName
End Function

Function GetDataAsDictArr(targetWs As Worksheet, headerRow As Integer, startCol As Integer, endCol As Integer, _
                            Optional lastRowCol As Integer = 1, Optional filterCol As Variant = 0, Optional filterVal As Variant = "") As Object
    Dim dict                As Object
    Dim header           As Variant
    Dim i       As Long, r      As Long
    Dim lastRow         As Long
    Dim valueList        As Object
    Dim tempArr        As Variant
    Dim rowCount     As Long

    ' Dictionary 생성
    Set dict = CreateObject("Scripting.Dictionary")

    ' 마지막 행 찾기 (기본 1열, 입력받으면 해당 컬럼 사용)
    lastRow = targetWs.Cells(targetWs.Rows.count, lastRowCol).End(xlUp).Row

    ' 컬럼별 리스트 저장용 Dictionary 생성
    Set valueList = CreateObject("Scripting.Dictionary")

    ' 초기화: 각 컬럼별 리스트 저장할 Dictionary에 빈 ArrayList 추가
    For i = startCol To endCol
        header = Trim(targetWs.Cells(headerRow, i).value)
        Set valueList(header) = CreateObject("System.Collections.ArrayList")
    Next i

    ' 데이터 읽기 및 필터링 적용
    For r = headerRow + 1 To lastRow
        ' 필터링 조건 확인 (filterCol이 1 이상일 때만 체크)
        If filterCol > 0 Then
            If targetWs.Cells(r, filterCol).value <> filterVal Then GoTo nextRow
        End If

        ' 각 컬럼별 데이터 저장
        For i = startCol To endCol
            header = Trim(targetWs.Cells(headerRow, i).value)
            valueList(header).Add targetWs.Cells(r, i).value
        Next i

nextRow:
    Next r

    ' Dictionary에 1부터 시작하는 배열 저장
    For Each header In valueList.keys
        rowCount = valueList(header).count
        If rowCount < 1 Then GoTo SkipGetData:      '해당 연도 데이터가 없을 경우 데이터 추출 skip
        ReDim tempArr(1 To rowCount) ' 1부터 시작하는 배열 선언

        ' ArrayList 데이터를 tempArr에 복사 (1부터 시작)
        For i = 1 To rowCount
            tempArr(i) = valueList(header).Item(i - 1) ' ArrayList는 0부터 시작
        Next i

        ' 변환된 배열을 Dictionary에 저장
        dict(header) = tempArr
    Next header

SkipGetData:
    ' 결과 반환
    Set GetDataAsDictArr = dict
End Function

Function ExtractNumericOnly(ByVal inputText As Variant) As Long
'숫자만 추출
    Dim regEx               As Object
    Dim cleanedText     As String
    Set regEx = CreateObject("VBScript.RegExp")
    
    ' 빈값 0 처리
    If IsNull(inputText) Or Trim(CStr(inputText)) = "" Then
        ExtractNumericOnly = 0
        Exit Function
    End If
    
    ' 숫자, 소수점, 음수 기호 외는 제거
    With regEx
        .Global = True
        .IgnoreCase = True
        .Pattern = "[^0-9.\-]"
    End With

    cleanedText = regEx.Replace(CStr(inputText), "")
    
    ' 유효하지 않은 숫자 문자열은 0 처리
    If cleanedText = "" Or cleanedText = "-" Or cleanedText = "." Or cleanedText = "-." Then
        ExtractNumericOnly = 0
    Else
        ExtractNumericOnly = CLng(cleanedText)
    End If
End Function
Function FindCellInRange(ByVal rng As Range, ByVal keyword As String, Optional ByVal matchWhole As Boolean = False) As Range
'특정 범위에서 공백 제거 후 키워드로 검색하여 셀주소 반환
Dim c As Range
    Dim key As String, txt As String
    
    ' 유효성 검사
    If rng Is Nothing Then Exit Function
    If Len(Trim$(keyword)) = 0 Then Exit Function
    
    ' 키워드의 모든 공백 제거
    key = Replace(Trim$(keyword), " ", "")
    
    ' 범위 내 모든 셀 순회
    For Each c In rng.Cells
        If Not IsError(c.value) Then
            txt = CStr(c.value)
            
            ' 모든 공백류 제거 (NBSP, 전각, 개행, 탭 포함)
            txt = Replace(txt, Chr(160), "")
            txt = Replace(txt, ChrW(12288), "")
            txt = Replace(txt, vbTab, "")
            txt = Replace(txt, vbCr, "")
            txt = Replace(txt, vbLf, "")
            txt = Replace(txt, " ", "")
            
            ' 비교
            If matchWhole Then
                If StrComp(txt, key, vbTextCompare) = 0 Then
                    Set FindCellInRange = c
                    Exit Function
                End If
            Else
                If InStr(1, txt, key, vbTextCompare) > 0 Then
                    Set FindCellInRange = c
                    Exit Function
                End If
            End If
        End If
    Next c
    
    ' 못 찾으면 Nothing 반환
    Set FindCellInRange = Nothing
End Function
Function NextMergedStartRight(ByVal r As Range) As Range
' 기준 셀 r 가 속한 병합영역의 "오른쪽 이웃" 시작셀을 반환
    
    Dim ma As Range, nextCell As Range
    If r Is Nothing Then Exit Function
    Set ma = r.MergeArea                 ' r 이 병합이 아니면 자신(1x1) 반환
    
    ' 현재 병합영역의 맨 오른쪽 셀에서 한 칸 오른쪽
    Set nextCell = ma.Cells(1, ma.Columns.count).Offset(0, 1)
    If nextCell Is Nothing Then Exit Function
    If nextCell.MergeCells Then
        Set NextMergedStartRight = nextCell.MergeArea.Cells(1, 1)  '다음 병합의 좌상단(=S59 같은 위치)
    Else
        Set NextMergedStartRight = nextCell                         '병합이 아니면 그 셀 자체
    End If
End Function

                        '--------------------------------------------

                    End If


                    '================================================
                    ' 17. 삭제 대상 급여 / 공제 항목 제거
                    '================================================
                    On Error Resume Next


                    If UBound(DelKeyWords) >= _
                        LBound(DelKeyWords) Then


                        For d = _
                            UBound(DelKeyWords) _
                            To LBound(DelKeyWords) _
                            Step -1


                            If Trim( _
                                CStr(DelKeyWords(d)) _
                            ) <> "" Then


                                Set headerDict = _
                                    GetHeaderAsDictionary( _
                                        targetWs, _
                                        1 _
                                    )


                                If headerDict.Exists( _
                                    DelKeyWords(d) _
                                ) Then


                                    .Columns( _
                                        headerDict( _
                                            DelKeyWords(d) _
                                        ) _
                                    ).Delete


                                End If

                            End If

                        Next d

                    End If


                    Err.Clear
                    On Error GoTo EH


                    '================================================
                    ' 18. 사번 컬럼 보정
                    '
                    ' oneRow 결과에 "사번"이 없고
                    ' "주민번호"가 존재하면
                    '
                    ' A열에 "사번"을 새로 만들고
                    ' 주민번호 값을 그대로 복사
                    '================================================

                    Set finalHeaderDict = _
                        GetHeaderAsDictionary( _
                            targetWs, _
                            1 _
                        )


                    If Not finalHeaderDict.Exists("사번") Then

                        residentHeaderName = ""


                        '--------------------------------------------
                        ' Config에 정의된 주민번호 헤더 우선 사용
                        '--------------------------------------------
                        If Trim(CStr(resiIDName)) <> "" Then

                            If finalHeaderDict.Exists(resiIDName) Then

                                residentHeaderName = resiIDName

                            End If

                        End If


                        '--------------------------------------------
                        ' 못 찾으면 일반적인 "주민번호" 검색
                        '--------------------------------------------
                        If residentHeaderName = "" Then

                            If finalHeaderDict.Exists("주민번호") Then

                                residentHeaderName = "주민번호"

                            End If

                        End If


                        '--------------------------------------------
                        ' "주민등록번호" 형식도 대응
                        '--------------------------------------------
                        If residentHeaderName = "" Then

                            If finalHeaderDict.Exists("주민등록번호") Then

                                residentHeaderName = "주민등록번호"

                            End If

                        End If


                        '--------------------------------------------
                        ' 주민번호 컬럼 존재 시 사번 생성
                        '--------------------------------------------
                        If residentHeaderName <> "" Then

                            residentCol = _
                                CLng( _
                                    finalHeaderDict( _
                                        residentHeaderName _
                                    ) _
                                )


                            finalLastRow = .Cells( _
                                .Rows.count, _
                                residentCol _
                            ).End(xlUp).Row


                            '----------------------------------------
                            ' 맨 앞 A열 삽입
                            '----------------------------------------
                            .Columns(1).Insert _
                                Shift:=xlToRight, _
                                CopyOrigin:=xlFormatFromLeftOrAbove


                            '----------------------------------------
                            ' A1 헤더
                            '----------------------------------------
                            .Cells(1, 1).value = "사번"


                            '
                            ' A열 삽입 후 기존 주민번호 컬럼도
                            ' 한 칸 오른쪽으로 밀림
                            '
                            residentCol = residentCol + 1


                            '----------------------------------------
                            ' 주민번호를 사번으로 복사
                            '----------------------------------------
                            If finalLastRow >= 2 Then

                                .Range( _
                                    .Cells(2, 1), _
                                    .Cells(finalLastRow, 1) _
                                ).value = _
                                .Range( _
                                    .Cells(2, residentCol), _
                                    .Cells(finalLastRow, residentCol) _
                                ).value

                            End If


                            '----------------------------------------
                            ' 사번은 텍스트 형식
                            '----------------------------------------
                            .Columns(1).NumberFormat = "@"


                            Debug.Print _
                                "사번 자동 생성 완료 : " & _
                                rawWbName & _
                                " / " & _
                                targetWs.Name & _
                                " / 기준=" & _
                                residentHeaderName


                        Else

                            '
                            ' 주민번호 컬럼도 없으면
                            ' 조용히 넘어가지 않고 오류 표시
                            '
                            Err.Raise _
                                vbObjectError + 3030, _
                                "oneRow_payrollConvert", _
                                "사번 컬럼이 없으며 주민번호 컬럼도 찾을 수 없습니다." & _
                                vbCrLf & _
                                "파일 : " & rawWbName & _
                                vbCrLf & _
                                "시트 : " & targetWs.Name

                        End If

                    End If


                    '================================================
                    ' 19. 서식 정리
                    '================================================
                    .UsedRange.Borders.LineStyle = xlNone


                    If Application.WorksheetFunction.CountA( _
                        .Cells _
                    ) > 0 Then


                        .Cells(1, 1) _
                            .CurrentRegion _
                            .Borders.LineStyle = xlContinuous


                        .Cells(1, 1) _
                            .CurrentRegion _
                            .Columns.AutoFit

                    End If


                End With


NEXT_SHEET:

            Next targetWs


            '================================================
            ' 20. 변환 결과 저장
            '================================================
            wsCount = targetWb.Worksheets.count


            If wsCount > 1 Then

                If InStr(title, "년") > 0 Then

                    title = Left( _
                        title, _
                        InStr(title, "년") _
                    )

                End If

            End If


            fiteredWbName = Replace( _
                rawWbName, _
                "[", _
                "(" _
            )


            fiteredWbName = Replace( _
                fiteredWbName, _
                "]", _
                ")" _
            )


            targetWb.SaveAs _
                fileName:= _
                    processPayrollPath & "\" & _
                    corpName & "_" & _
                    title & _
                    " oneRow 대장_" & _
                    fiteredWbName


            targetWb.Close _
                SaveChanges:=False


            Set targetWs = Nothing
            Set targetWb = Nothing

        End If


        '====================================================
        ' 다음 파일
        '====================================================
        rawWbName = Dir

    Loop


CLEAN_EXIT:

    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.DisplayStatusBar = True
    Application.StatusBar = False

    Set targetWs = Nothing
    Set targetWb = Nothing

    Exit Sub


EH:

    '========================================================
    ' 원래 오류 정보 저장
    '========================================================
    savedErrNumber = Err.Number
    savedErrDescription = Err.Description
    savedErrSource = Err.Source

    errorFileName = rawWbName
    errorSheetName = ""


    '========================================================
    ' 시트명 안전하게 확보
    '========================================================
    On Error Resume Next

    If Not targetWs Is Nothing Then
        errorSheetName = targetWs.Name
    End If

    Err.Clear


    '========================================================
    ' Excel 상태 복원
    '========================================================
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.DisplayStatusBar = True
    Application.StatusBar = False


    '========================================================
    ' Workbook 닫기
    '========================================================
    If Not targetWb Is Nothing Then

        targetWb.Close _
            SaveChanges:=False

    End If

    Err.Clear

    Set targetWs = Nothing
    Set targetWb = Nothing

    On Error GoTo 0


    '========================================================
    ' 원래 오류 출력
    '========================================================
    MsgBox _
        "Module3 급여대장 변환 중 오류가 발생했습니다." & _
        vbCrLf & vbCrLf & _
        "파일 : " & errorFileName & vbCrLf & _
        "시트 : " & errorSheetName & vbCrLf & _
        "오류번호 : " & savedErrNumber & vbCrLf & _
        "오류위치 : " & savedErrSource & vbCrLf & _
        "내용 : " & savedErrDescription, _
        vbCritical, _
        "Module3 오류"

    Exit Sub

End Sub

Sub corpInfoExtract(corpName As String, year As Variant)
'1. "OneRow 급여대장" 파일 "지급/공제항목" 추출하여 기본정보 파일에 입력
    Application.ScreenUpdating = False
    Application.DisplayStatusBar = False
    Application.DisplayAlerts = False
    
'1-1. "OneRow급여대장" 파일에서 "지급/공제항목" 추출
    Dim payListDict                        As Object
    Dim subtractListDict                As Object
    Dim oneRowWbNamePattern       As String
    Dim oneRowWbName                   As String
    Dim oneRowWb                            As Workbook
    Dim oneRowWs                             As Worksheet
    
    Set payListDict = CreateObject("Scripting.Dictionary")
    Set subtractListDict = CreateObject("Scripting.Dictionary")
    
    yearName = year & "년"
    oneRowWbNamePattern = "*" & yearName & "*oneRow*"
    oneRowWbName = Dir(processPayrollPath & "\*.xls*")
   
    Do While oneRowWbName <> ""
   
        If oneRowWbName Like oneRowWbNamePattern Then
            Set oneRowWb = Workbooks.Open(processPayrollPath & "\" & oneRowWbName)

            For Each oneRowWs In oneRowWb.Sheets
               
                Dim oneRowHeaderColDict             As Object
                Dim oneRowWsEndRow                  As Long
                Dim payDataFirstCol                         As Integer
                Dim payDataLastCol                         As Integer
                Dim subtractDataLastCol                 As Integer
                Dim payHeaderRng                          As Range
                Dim subtractHeaderRng                   As Range
                Dim cell                                             As Range
                
                With oneRowWs
                    Set oneRowHeaderColDict = GetHeaderAsDictionary(oneRowWs, 1)
                    oneRowWsEndRow = .Cells(Rows.count, 1).End(xlUp).Row
                    payDataFirstCol = oneRowHeaderColDict(KeyWords(3, 1))
                    payDataLastCol = oneRowHeaderColDict(KeyWords(4, 1))
                    subtractDataLastCol = oneRowHeaderColDict(KeyWords(7, 1))
                                                            
                    Set payHeaderRng = .Range(.Cells(1, payDataFirstCol), .Cells(1, payDataLastCol - 1))                          '지급항목 헤더
                    Set subtractHeaderRng = .Range(.Cells(1, payDataLastCol + 1), .Cells(1, subtractDataLastCol - 2))     '공제항목 헤더

                    
                    ' "지급/공제항목" 딕셔너리 키에 저장(중복 제거)
                    For Each cell In payHeaderRng
                        If Not payListDict.Exists(cell.value) Then
                            payListDict.Add cell.value, year             '당해 지급항목
                        End If
                        If Not duplicPayListDict.Exists(cell.value) Then
                            duplicPayListDict.Add cell.value, True   '누적 지급항목
                        End If
                    Next cell
                    For Each cell In subtractHeaderRng
                        If Not subtractListDict.Exists(cell.value) Then
                            subtractListDict.Add cell.value, year       '당해 공제항목
                        End If
                        If Not duplicSubtractListDict.Exists(cell.value) Then
                            duplicSubtractListDict.Add cell.value, True      '누적 지급항목
                        End If
                    Next cell

                End With
                
            Next oneRowWs
        End If
        
        If Not oneRowWb Is Nothing Then
            oneRowWb.Close SaveChanges:=False
            Set oneRowWb = Nothing
        End If
        oneRowWbName = Dir
    Loop
        
    
    '기초정보 파일 "사업장정보" 시트에 "지급/공제항목" 입력
    Dim payListCol                                  As Integer
    Dim subtractListCol                          As Integer
    Dim duplicPayListCol                        As Integer
    Dim duplicSubtractListCol                 As Integer
    Dim payListEndRow                          As Integer
    Dim subtractListEndRow                  As Integer

    With corpInfoWs
        payListCol = corpInfoWsHeaderColDict("연도별 지급항목")
        subtractListCol = corpInfoWsHeaderColDict("연도별 공제항목")
        duplicPayListCol = corpInfoWsHeaderColDict("지급항목")
        duplicSubtractListCol = corpInfoWsHeaderColDict("공제항목")
        payListEndRow = .Cells(Rows.count, payListCol).End(xlUp).Row
        subtractListEndRow = .Cells(Rows.count, subtractListCol).End(xlUp).Row
    
        Dim inputRow        As Integer
        Dim key                   As Variant
        
        '연도별 지급항목 입력
        inputRow = payListEndRow + 1
        For Each key In payListDict.keys
            .Cells(inputRow, payListCol).value = key
            .Cells(inputRow, payListCol + 3).value = payListDict(key)
            inputRow = inputRow + 1
        Next key
        '연도별 공제항목 입력
        inputRow = subtractListEndRow + 1
        For Each key In subtractListDict.keys
            .Cells(inputRow, subtractListCol).value = key
            .Cells(inputRow, subtractListCol + 1).value = subtractListDict(key)
            inputRow = inputRow + 1
        Next key
    End With

    Set oneRowHeaderColDict = Nothing
    Set payListDict = Nothing
    Set subtractListDict = Nothing
    
End Sub
Sub duplicListInput(corpName As String)
' 지급/공제항목 연도별 누적 리스트 입력

    Dim duplicPayListCol                       As Integer
    Dim duplicSubtractListCol               As Integer
    Dim duplicPayListEndRow               As Integer
    Dim duplicSubtractListEndRow      As Integer
    Dim lastRow                                     As Integer
    Dim inputRow                                 As Integer
    Dim key                                            As Variant
  
    With corpInfoWs
        duplicPayListCol = corpInfoWsHeaderColDict("지급항목")
        duplicSubtractListCol = corpInfoWsHeaderColDict("공제항목")
        duplicPayListEndRow = .Cells(Rows.count, duplicPayListCol).End(xlUp).Row
        duplicSubtractListEndRow = .Cells(Rows.count, duplicSubtractListCol).End(xlUp).Row
        
        lastRow = Application.WorksheetFunction.Max(duplicPayListEndRow, duplicSubtractListEndRow, 6)
        .Range(.Cells(6, duplicPayListCol), .Cells(lastRow, duplicSubtractListCol)).Delete Shift:=xlUp
        .Range(.Cells(5, duplicPayListCol), .Cells(5, duplicSubtractListCol)).ClearContents
        
        inputRow = 5         '누적 지급항목 입력
        For Each key In duplicPayListDict.keys
            .Cells(inputRow, duplicPayListCol).value = key
            inputRow = inputRow + 1
        Next key
        inputRow = 5         '누적 공제항목 입력
        For Each key In duplicSubtractListDict.keys
            .Cells(inputRow, duplicSubtractListCol).value = key
            inputRow = inputRow + 1
        Next key

        ' "지급항목" 체크란 복붙 & "지급항목", "공제항목" font 정리
        Dim checkCopyRng                 As Range
        Dim checkPasteRng                 As Range
        
        duplicPayListEndRow = .Cells(Rows.count, duplicPayListCol).End(xlUp).Row
        duplicSubtractListEndRow = .Cells(Rows.count, duplicSubtractListCol).End(xlUp).Row
        Set checkCopyRng = .Range(.Cells(5, duplicPayListCol + 1), .Cells(5, duplicPayListCol + 2))
        Set checkPasteRng = .Range(.Cells(5, duplicPayListCol + 1), .Cells(duplicPayListEndRow, duplicPayListCol + 2))
        
        checkCopyRng.Copy
        checkPasteRng.PasteSpecial Paste:=xlPasteAll
        Application.CutCopyMode = False
        
        With .Range(.Cells(4, duplicPayListCol), .Cells(duplicPayListEndRow, duplicPayListCol + 2))
            .Borders.LineStyle = xlContinuous
            .Font.Size = 10
            .Font.Name = "맑은 고딕"
            .Font.Bold = True
            .HorizontalAlignment = xlCenter
        End With
        With .Range(.Cells(4, duplicSubtractListCol), .Cells(duplicSubtractListEndRow, duplicSubtractListCol))
            .Borders.LineStyle = xlContinuous
            .Font.Size = 10
            .Font.Name = "맑은 고딕"
            .Font.Bold = True
            .HorizontalAlignment = xlCenter
        End With
        
        .Activate
        .Cells(1, 1).Select
    End With
    
    Set duplicPayListDict = Nothing
    Set duplicSubtractListDict = Nothing
    Set corpInfoWsHeaderColDict = Nothing

    basicInfoWb.SaveAs fileName:=refundTargetPath & "\" & corpName & "_기초정보(1차).xlsx"
    basicInfoWb.Close SaveChanges:=False

    
    Application.ScreenUpdating = True
    Application.DisplayStatusBar = True
    Application.DisplayAlerts = True
    
End Sub

Sub basic2ndWbUpdate(corpName)
'[기초정보(2차)] 파일 데이터 업데이트 (비과세 항목, 지급항목(정액/수당검토))
'1-1. [수당등록]시트에서 비과세 항목명 추출 >> [수당조정(비과세)]시트 입력
    Set basic2ndWb = Workbooks.Open(refundTargetPath & "\" & corpName & "_기초정보(2차).xlsx")

    Dim lastCol                           As Integer
    Dim lastRow                         As Long
    Dim tfPayNameArr()             As Variant
    Dim headerColDict               As Object
    
    Set headerColDict = GetHeaderAsDictionary(payAdjustWs, 4)
    
    With payAdjustWs
        lastCol = .Cells(4, Columns.count).End(xlToLeft).Column
        If lastCol > headerColDict("과세차감총액") Then: .Range(.Cells(2, headerColDict("과세차감총액") + 1), .Cells(4, lastCol)).Clear       '기존 입력 데이터 삭제(재수행 고려)
    End With
    
    With payRegisterWs       '추출
        lastRow = .Cells(Rows.count, 1).End(xlUp).Row
        ReDim tfPayNameArr(1 To lastRow - 3, 3)
        For i = 1 To UBound(tfPayNameArr, 1)
            tfPayNameArr(i, 1) = .Cells(i + 3, 5).value     '코드
            tfPayNameArr(i, 2) = .Cells(i + 3, 4).value     '근로소득세부
            tfPayNameArr(i, 3) = .Cells(i + 3, 1).value     '수당명
        Next i
    End With
    
    With payAdjustWs         '[수당조정(비과세)]시트에 입력
        lastCol = .Cells(4, Columns.count).End(xlToLeft).Column
        .Cells(2, lastCol + 1).Resize(3, UBound(tfPayNameArr, 1)).value = Application.WorksheetFunction.Transpose(tfPayNameArr)
    End With
    
'1-2. [사업장정보]시트 지급항목(정액/수당검토) O표시 >> 연도별 지급항목에 복사
    Dim corpInfoWsHeaderColDict       As Object
    Dim payListCol                                 As Integer
    Dim updatePayListCol                      As Integer
    Dim payListDict                                As Object
    Dim tempArr                                     As Variant
    Dim resultArr                                    As Variant
    
    Set corpInfoWs = basic2ndWb.Sheets("사업장정보")
    Set corpInfoWsHeaderColDict = GetHeaderAsDictionary(corpInfoWs, 4)
    
    With corpInfoWs
        payListCol = corpInfoWsHeaderColDict("지급항목")
        updatePayListCol = corpInfoWsHeaderColDict("연도별 지급항목")
        
        '지급항목 구분 정보 저장(key:지급항목, value:(정액급여, 수당검토))
        lastRow = .Cells(Rows.count, payListCol).End(xlUp).Row

        Set payListDict = CreateObject("Scripting.Dictionary")
        
        For i = 1 To UBound(tempArr, 1)
            payListDict(tempArr(i, 1)) = Array(tempArr(i, 2), tempArr(i, 3))
        Next i
        
        '연도별 지급항보 정보에 입력
        lastRow = .Cells(Rows.count, updatePayListCol).End(xlUp).Row
        tempArr = .Range(.Cells(5, updatePayListCol), .Cells(lastRow, updatePayListCol)).value

            If payListDict.Exists(tempArr(i, 1)) Then
                resultArr(i, 1) = payListDict(tempArr(i, 1))(1)     '정액급여 값
                resultArr(i, 2) = payListDict(tempArr(i, 1))(2)     '수당체크 값
            End If
        Next i
        
'        For i = 1 To UBound(resultArr, 1)
'            Debug.Print resultArr(i, 1)
'        Next i
        
        .Range(.Cells(5, updatePayListCol + 1), .Cells(lastRow, updatePayListCol + 2)).value = resultArr
               
    End With

    Erase resultArr
    Set payListDict = Nothing

'2-2. 월별 [oneRow] 파일에서 데이터 추출 후 [급상여현황] 파일 입력

    Dim rawWbName                      As String

    If Dir(totalPayPath, vbDirectory) = "" Then MkDir (totalPayPath)

    targetWbNamePattern = "*" & yearName & "*oneRow*"

            For Each targetWs In targetWb.Sheets

                Dim totalPayWsHeaderColDict        As Object

                For r = 2 To UBound(payDataArr, 1)         '[oneRow]파일 데이터 부분 반복(첫 행은 헤더)
                    'payDataArr 헤더와 일치하면 outputDataArr1입력

                        key = totalPayWs.Cells(1, c).value
                        For j = 1 To UBound(payDataArr, 2)

                            End If

                    For c = payDataEndCol + 1 To subtractDataEndCol
                        key = totalPayWs.Cells(1, c).value

'2-3. [급상여현황] 파일 가공
    Dim incomeTaxKey           As String
    Dim fixedPayValue           As Long
    Dim checkPayValue          As Long
    Dim a                                As Integer
    Dim empKey                     As String
    Dim targetDict                  As Object
    Dim summaryDict             As Object
    Dim targetArr                   As Variant
    Dim resiNum                     As String
    
    With totalPayWs

        Set summaryDict = CreateObject("Scripting.Dictionary")
        
        Set totalPayWsHeaderColDict = GetHeaderAsDictionary(totalPayWs, 1)
        totalPayWsEndCol = .Cells(1, Columns.count).End(xlToLeft).Column
        .Columns("a").NumberFormat = "@"
        
        For r = 2 To totalPayWsEndRow
'            '사번이 4자리 수 미만일 경우 앞에 0 붙여서 4자리 수로 맞춤
            empKey = Trim(CStr(.Cells(r, 1).value))
'            If empKey <> "" Then
'                If Len(empKey) < 4 Then
'                    .Cells(r, 1).Value = Right("0000" & empKey, 4)
'                End If
'            End If

            empKey = Trim(CStr(.Cells(r, 1).value))
            
            If empKey <> "" Then
                ' 사번을 그대로 다시 셀에 담되, 텍스트 형식으로 저장
                '.Cells(r, 1).NumberFormat = "@"
                .Cells(r, 1).value = empKey
            End If
            
            
            '엑셀 버전 아닐 경우 인원(사번)별 "주민등록번호" 입력(from원천징수추출)
            If Solution <> "엑셀" Then
                incomeTaxKey = CStr(.Cells(r, totalPayWsHeaderColDict(empNumName)).value)
                If incomeTaxDict.Exists(incomeTaxKey) Then
                     .Cells(r, totalPayWsHeaderColDict(resiIDName)).value = incomeTaxDict(incomeTaxKey)(resiIDName)
                End If
            End If
            '주민번호에 '-' 없을 경우 삽입
            resiNum = .Cells(r, totalPayWsHeaderColDict(resiIDName)).value
            If InStr(resiNum, "-") = 0 Then
                resiNum = Left(resiNum, 6) & "-" & Right(resiNum, 7)
            End If
            .Cells(r, totalPayWsHeaderColDict(resiIDName)).value = resiNum
                
            ' 월별 '정액급여합계' 계산
            fixedPayValue = 0
            For a = LBound(fixedPayListArr) To UBound(fixedPayListArr)
                If totalPayWsHeaderColDict.Exists(fixedPayListArr(a)) Then
                    fixedPayValue = fixedPayValue + .Cells(r, totalPayWsHeaderColDict(fixedPayListArr(a))).value
                End If
            Next a
            .Cells(r, totalPayWsHeaderColDict("정액급여합계")) = fixedPayValue
            
            ' 월별 수당검토 항목들의 합이 0보다 크고, [정액급여합계]가 210만원 이하인 달 행 노란색 표기 & 해당 사번 딕셔너리 추출
            checkPayValue = 0
            For a = LBound(checkPayListArr) To UBound(checkPayListArr)
                If totalPayWsHeaderColDict.Exists(checkPayListArr(a)) Then
                    checkPayValue = checkPayValue + .Cells(r, totalPayWsHeaderColDict(checkPayListArr(a))).value
                End If
            Next a

            If checkPayValue > 0 And .Cells(r, totalPayWsHeaderColDict("정액급여합계")) <= 2100000 Then
               .Range(.Cells(r, totalPayWsHeaderColDict("지급년월")), .Cells(r, totalPayWsHeaderColDict("정액급여합계"))).Interior.Color = RGB(255, 230, 153)  '노란색
                
                If Not IsEmpty(empKey) And Trim(empKey) <> "" Then
                    If Not targetDict.Exists(empKey) Then
                        targetDict.Add empKey, True
                    End If
                End If
            End If
                   
                   
            Dim empName                      As String
            Dim resignDateCol                As Integer
            Dim summaryRow                 As Long
            Dim sumKey                          As Variant
            Dim preIncomeKey                As String
            Dim netName                         As String
            Dim cellValue                         As String
            Dim openParenPos                As Integer
            Dim closeParenPos                As Integer
            Dim front8ResiNum               As String
            Dim personalKey                   As Variant

            empNum = CStr(.Cells(r, 1).value)

            resiNum = .Cells(r, totalPayWsHeaderColDict(resiIDName)).value
            yearMonthCol = totalPayWsHeaderColDict("지급년월")
            payTypeCol = totalPayWsHeaderColDict("급여구분")
            resignDateCol = yearMonthCol - 1   '"퇴사일자" 컬럼
            payType = .Cells(r, payTypeCol).value
            
            sumKey = empNum & payType
            
            If Not summaryDict.Exists(sumKey) Then
                Dim tempArr()           As Variant
                ReDim tempArr(1 To totalPayWsEndCol)
                For c = 1 To yearMonthCol - 2           ' 개인별 정보 : "사원번호" ~ "입사일"
                    tempArr(c) = .Cells(r, c).value
                Next c
                tempArr(yearMonthCol) = yearName & "합계"      ' "지급년월"
                tempArr(payTypeCol) = .Cells(r, 7).value              ' "급여구분"
                summaryDict.Add sumKey, tempArr
            End If
            
            tempArr = summaryDict(sumKey)
            
            If .Cells(r, resignDateCol).value <> "" Then            '"퇴사일"은 인원별 첫달 데이터에 없을 수 있으므로 별도 추가
                tempArr(resignDateCol) = .Cells(r, resignDateCol).value
            End If

                If IsNumeric(.Cells(r, c).value) Then
                    tempArr(c) = tempArr(c) + .Cells(r, c).value
                End If
            Next c
            
            summaryDict(sumKey) = tempArr
            
        Next r
        
        
        ' 인원(사번or주번)&급여구분별 합계행 입력
        summaryRow = totalPayWsEndRow + 1
        For Each sumKey In summaryDict.keys
            tempArr = summaryDict(sumKey)

                .Cells(summaryRow, c).value = tempArr(c)
                .Range(.Cells(summaryRow, 1), .Cells(summaryRow, c)).Interior.Color = RGB(169, 208, 142)  '합계행 > 녹색

            Else
                incomeTaxKey = .Cells(summaryRow, totalPayWsHeaderColDict(empNumName)).value

            If Not preIncomeDict Is Nothing Then
                If preIncomeDict.Exists(preIncomeKey) Then
                    .Cells(summaryRow, totalPayWsHeaderColDict("전년도과세소득_총계")).value = preIncomeDict(preIncomeKey)
                End If

            '인원별 "직업분류코드" 입력 (key : 이름 + 주민번호 앞 8자리)
            cellValue = .Cells(summaryRow, 2).value
            openParenPos = InStr(cellValue, "(")
            closeParenPos = InStrRev(cellValue, ")")
            If openParenPos > 0 And closeParenPos > 0 And closeParenPos > openParenPos Then
                netName = Replace(cellValue, Mid(cellValue, openParenPos, closeParenPos - openParenPos + 1), "")          '"이름"에서 괄호 부분 제거
            Else
                netName = cellValue
            End If
            front8ResiNum = Left(.Cells(summaryRow, totalPayWsHeaderColDict(resiIDName)).value, 8)    ' "주민번호" 앞 8자리
            jobCodeKey = netName & front8ResiNum
            
            If Not jobCodeDict Is Nothing Then
                If jobCodeDict.Exists(jobCodeKey) Then
                    .Cells(summaryRow, totalPayWsHeaderColDict("직업분류 코드")).value = jobCodeDict(jobCodeKey)(1)
                    .Cells(summaryRow, totalPayWsHeaderColDict("직업분류 항목")).value = jobCodeDict(jobCodeKey)(2)
                End If
            End If
            
            summaryRow = summaryRow + 1
        Next sumKey
        
        
        '합계행 반복하여 셀 배경색 변경 및 해당 사번 추출
        Dim resiNumCol                  As Integer
        Dim resiNumCheck             As String
        Dim payTypeCheck              As String
        Dim sameEmpDict               As Dictionary
        Dim overPayDict               As Dictionary
        Dim exceptDict                As Dictionary
    
        Set sameEmpDict = New Dictionary
        Set overPayDict = New Dictionary
        Set exceptDict = New Dictionary
        resiNumCol = totalPayWsHeaderColDict(resiIDName)
        
        For r = totalPayWsEndRow + 1 To summaryRow - 1

            resiNumCheck = .Cells(r, resiNumCol).value
            payTypeCheck = .Cells(r, payTypeCol).value
            
            If Application.WorksheetFunction.CountIfs( _
            .Range(.Cells(totalPayWsEndRow + 1, resiNumCol), .Cells(summaryRow, resiNumCol)), resiNumCheck, _
            .Range(.Cells(totalPayWsEndRow + 1, payTypeCol), .Cells(summaryRow, payTypeCol)), payTypeCheck) > 1 Then
            
                .Cells(r, resiNumCol).Interior.Color = RGB(142, 169, 219)   '"주민등록번호" 컬럼 푸른색
                '.Cells(r, 1).Interior.Color = RGB(142, 169, 219)                    '"사번" 컬럼 푸른색

                End If

            If .Cells(r, totalPayWsHeaderColDict("전년도과세소득_총계")).value > 30000000 Then
                '.Cells(r, 1).Interior.Color = RGB(255, 199, 206)                    '"사번" 컬럼 붉은색
                .Cells(r, totalPayWsHeaderColDict("전년도과세소득_총계")).Interior.Color = RGB(255, 199, 206)
                If Not overPayDict.Exists(.Cells(r, 1).value) Then
                    overPayDict.Add .Cells(r, 1).value, True                             'overPayDict 키 추가
                End If

            '대상 제외인 경우 처리
            If InStr(.Cells(r, totalPayWsHeaderColDict("직업분류 코드")).value, "아님") > 0 Then '.Cells(r, 1).Interior.Color = RGB(255, 199, 206)
                If Not exceptDict.Exists(.Cells(r, 1).value) Then

            If .Cells(r, totalPayWsHeaderColDict("직업분류 코드")).value = "" Then .Cells(r, totalPayWsHeaderColDict("직업분류 코드")).Interior.Color = RGB(255, 190, 140)

            If .Cells(r, totalPayWsHeaderColDict("과세소득_총계")).value = "" Then .Cells(r, totalPayWsHeaderColDict("과세소득_총계")).Interior.Color = RGB(255, 190, 140)

        .Range(.Cells(2, totalPayWsHeaderColDict("정액급여합계") + 4), .Cells(totalPayWsEndRow, totalPayWsEndCol)).NumberFormat = "#,###"
        .Range(.Cells(2, 4), .Cells(totalPayWsEndRow, 5)).NumberFormat = "yyyy-mm-dd"

        Dim keyExist                  As Boolean

            Set targetCell = .Cells(r, 1)
            
            keyExist = targetDict.Exists(CStr(targetCell.value))
            If keyExist Then targetCell.Interior.Color = RGB(255, 230, 153)    '환급월이 1개월이라도 있을 경우 '사번' 노란색

            If keyExist Then targetCell.Interior.Color = RGB(255, 199, 206)   '직전년도 과세소득 3천만원 이상인 경우 '사번' 붉은색

'2-4. [비대상자] 시트 생성
            '비대상자만 보이기 (흰색:비대상, 붉은색:소득초과, 푸른색:재입사, 녹색:비대상합계액)

            If .Cells(r, 1).Interior.Color = RGB(255, 255, 255) Or .Cells(r, 1).Interior.Color = RGB(255, 199, 206) Or _
                .Cells(r, 1).Interior.Color = RGB(142, 169, 219) Or .Cells(r, 1).Interior.Color = RGB(169, 208, 142) Then
                .Rows(r).Hidden = False

            End If

        Dim totalPayNonTargetWs          As Worksheet

        With totalPayNonTargetWs
            .Name = "비대상자"
            .Range("A1").PasteSpecial Paste:=xlPasteAll
            Application.CutCopyMode = False
            .Cells(2, totalPayWsHeaderColDict("급여구분") + 1).Select
            ActiveWindow.FreezePanes = True
            .UsedRange.Columns.AutoFit
        End With

    End With

        
'2-5. [대상자], [대상자(작업)] 시트 생성

    Dim totalPayTargetCorrectWs    As Worksheet

    With totalPayWs
        .Range(.Cells(1, 1), .Cells(totalPayWsEndRow, totalPayWsEndCol)).AutoFilter Field:=1, _
                Criteria1:=RGB(255, 230, 153), Operator:=xlFilterCellColor      '"사번" 컬럼 노란색인 경우만 필터(대상자)

       .Cells(2, totalPayWsHeaderColDict("급여구분") + 1).Select

        Columns(correctWsColDict(paySumName)).Insert Shift:=xlToRight, CopyOrigin:=xlFormatFromLeftOrAbove

                Application.WorksheetFunction.Sum(.Range(.Cells(r, checkPayStartCol), .Cells(r, checkPayEndCol)))   '수당체크 항목 합계

            Dim empNumResiId                    As String
            Dim preRowEmpNumResiId       As String
            Dim isSumRow                            As Boolean
            Dim empStartRow                       As Long
            Dim empEndRow                        As Long
            
            '인원별 합계행에 합계 급액 수식 입력
            empNumResiId = CStr(.Cells(r, correctWsColDict(empNumName)).value & .Cells(r, correctWsColDict(resiIDName)).value)    '당행 키값
            preRowEmpNumResiId = CStr(.Cells(r - 1, correctWsColDict(empNumName)).value & .Cells(r - 1, correctWsColDict(resiIDName)).value)  '직전행 키값
            isSumRow = InStr(.Cells(r, correctWsColDict("지급년월")).value, "합계") > 0
            
            If Not isSumRow And empNumResiId <> preRowEmpNumResiId Then empStartRow = r     '인원별 시작행

                empEndRow = r - 1       '인원별 마지막행

                .Range(.Cells(r, checkPayStartCol), .Cells(r, correctWsColDict("야간연장수당(비과세)"))).FormulaR1C1 = _
                    "=SUMIF(R" & empStartRow & "C" & payTypeCol & ":R" & empEndRow & "C" & payTypeCol & ",R" & r & "C" & payTypeCol & ",R" & empStartRow & "C[0]:R" & empEndRow & "C[0])"
                    
            End If

        '"합계검증" 컬럼에 지급합계 금액 검증 수식 입력
        .Range(.Cells(2, correctWsColDict("합계검증")), .Cells(lastRow, correctWsColDict("합계검증"))).FormulaR1C1 = _
                          "=SUM(R[0]C" & totalPayStartCol & ":R[0]C" & totalPayEndCol & ")-R[0]C" & (totalPayEndCol + 2)

        Dim subtractStartCol            As Integer

        .Range(.Cells(1, subtractStartCol), .Cells(1, subtractEndCol)).EntireColumn.Hidden = True

        .Cells(2, totalPayStartCol).Select
        ActiveWindow.FreezePanes = True

                
    End With

    
    totalPayWb.SaveAs fileName:=totalPayPath & "\" & corpName & "_" & yearName & " 급상여현황.xlsx", FileFormat:=xlOpenXMLWorkbook
    totalPayWb.Close SaveChanges:=False
    basic2ndWb.Close SaveChanges:=False
    
    If Err.Number <> 0 Then
        Debug.Print "Error in totalPayCreate for year " & year & ": " & Err.Description
        Err.Clear
    End If
    

    
End Sub

Sub taxfreeExtract(corpName As String, phase As String, batchSize As Integer, year As Integer, batchCount As Integer)
'1. 연도별 원천징수영수증에서 비과세 소득 내역 추출 --> [기초정보 파일]-[원천징수추출(yy년)] 시트 작성
           
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.DisplayStatusBar = False
    
    Dim folderPath                          As String
    Dim fso                                      As Object
    Dim folder                                 As Object
    Dim file                                      As Object
    Dim fileList                                 As Object
    Dim totalBatchCount                 As Integer
    Dim startIdx                                As Integer
    Dim endIdx                                 As Integer
    
    Dim startTime                            As Single
    Dim endTime                             As Single
    Dim executionTime                    As Single
    
    startTime = Timer
    yearName = year & "년"
    

    If phase = "기존" Then
        folderPath = processPAYExlsx & "\" & yearName & "\"
    ElseIf phase = "수정" Then
        folderPath = outputPAYExlsx & "\" & yearName & "\"
    End If
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(folderPath) Then Exit Sub
    
    Set folder = fso.GetFolder(folderPath)
    Set fileList = CreateObject("System.Collections.ArrayList")         ' ※ 0부터 시작함!!!
    
    ' 파일 리스트 저장
    For Each file In folder.files
        If file.Name Like "*.xls*" And Not file.Name Like "~$*" Then        '임시 파일 제외
            fileList.Add file.Path
        End If
    Next file
    
    '총 파일 수 / 배치값 유효성 확인
    If batchSize <= 0 Then
        MsgBox "batchSize는 1 이상이어야 합니다.", vbExclamation, "Module2"
        Exit Sub
    End If

    If fileList.count = 0 Then
        MsgBox "처리할 원천징수영수증 Excel 파일이 없습니다." & vbCrLf & folderPath, vbInformation, "Module2"
        Exit Sub
    End If

    '총 파일 수에서 총 작업 횟수
    totalBatchCount = Application.WorksheetFunction.Ceiling(fileList.count / batchSize, 1)
    '작업 횟수에 따른 시작 파일, 종료 파일 인덱스
    startIdx = (batchCount - 1) * batchSize
    If startIdx > fileList.count - 1 Then
        MsgBox "현재 batchCount에 해당하는 처리 대상 파일이 없습니다." & vbCrLf & _
               "batchCount=" & batchCount & ", 파일수=" & fileList.count, vbInformation, "Module2"
        Exit Sub
    End If
    endIdx = Application.Min(startIdx + batchSize - 1, fileList.count - 1)
    
'    Debug.Print totalBatchCount
'    Debug.Print startIdx & " : " & fileList(startIdx)
'    Debug.Print endIdx & " : " & fileList(endIdx)
        
    
    Dim headerKeyDict          As Object
    Dim dataArr()                  As Variant
    Dim IsShExist                   As Boolean
    
    Set headerKeyDict = CreateObject("Scripting.Dictionary")
    
    '개인별 원천징수영수증 파일에서 데이터 추출
    Call extractPAYEData(fileList, startIdx, endIdx, headerKeyDict)
         
    
'1-2. 기초정보 파일에 [원천징수추출(yy년)] 시트 추가하여 비과세 내역 입력
    Dim rawFilePath                         As String
    Dim taxfreeExtractWsName       As String
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    If phase = "기존" Then
        rawFilePath = rawWbPath & "\" & corpName & "_기초정보.xlsx"
        processingFilePath = refundTargetPath & "\" & corpName & "_기초정보(1차).xlsx"
        taxfreeExtractWsName = "원천징수추출(" & yearName & ")"
    ElseIf phase = "수정" Then
        rawFilePath = inputPath & "\" & corpName & "_기초정보(3차).xlsx"
        processingFilePath = outputPath & "\" & corpName & "_기초정보(최종).xlsx"
        taxfreeExtractWsName = "원천징수추출(" & yearName & ")(수정)"
    End If
       
    '이미 '기초정보(1차)' 파일이 만들어져 있으면 1차 파일에 이어서 작업. 최초이면 '기초정보' 파일에서 작업
    If fso.FileExists(processingFilePath) Then
        Set basicInfoWb = Workbooks.Open(processingFilePath)
    Else
        Set basicInfoWb = Workbooks.Open(rawFilePath)
    End If
    
    IsShExist = False
    For Each taxfreeExtractWs In basicInfoWb.Sheets
        '동일한 시트명이 있으면 해당 시트 지정
        If taxfreeExtractWs.Name = taxfreeExtractWsName Then
            IsShExist = True
            Set taxfreeExtractWs = basicInfoWb.Sheets(taxfreeExtractWsName)
            Exit For
        End If
    Next taxfreeExtractWs
    
    '1단계일 경우에는 시트 지우고 새로 생성
    If IsShExist = True And batchCount = 1 Then
        taxfreeExtractWs.Delete
        Set taxfreeExtractWs = basicInfoWb.Sheets.Add(After:=basicInfoWb.Sheets(basicInfoWb.Sheets.count))
        taxfreeExtractWs.Name = taxfreeExtractWsName
    End If
    '시트가 없었으면 새로 생성
    If IsShExist = False Then
        Set taxfreeExtractWs = basicInfoWb.Sheets.Add(After:=basicInfoWb.Sheets(basicInfoWb.Sheets.count))
        taxfreeExtractWs.Name = taxfreeExtractWsName
    End If
               

    '연도별 원천징수영수증(yy년) 시트에 데이터 입력
    With taxfreeExtractWs
        Dim c                                       As Integer
        Dim extractWsEndCol             As Long
        Dim extractWsEndRow           As Long
        Dim headerName                     As String
        Dim key                                    As Variant
        
        extractWsEndCol = .Cells(1, Columns.count).End(xlToLeft).Column
        extractWsEndRow = .Cells(Rows.count, 1).End(xlUp).Row
        .Columns("B").NumberFormat = "@"        '사번 표기 문자로
                 
        ' A1셀이 비어 있으면 헤더 없음으로 판단
        If Trim(.Cells(1, 1).value) = "" Then
            extractWsEndCol = 0
        End If

        '헤더명 매칭하여 맞춰 각 데이터 배열 추가
        If extractWsEndCol > 0 Then
            For c = 1 To extractWsEndCol
                headerName = .Cells(1, c).value
                
                If headerKeyDict.Exists(headerName) Then
                    dataArr = headerKeyDict(headerName)
                    .Cells(extractWsEndRow + 1, c).Resize(UBound(dataArr), 1).value = Application.Transpose(dataArr)
                End If
            Next c
        End If
            
        '일치하는 헤더 없을 경우, 헤더명 추가하고 각 데이터 배열 추가
        For Each key In headerKeyDict.keys
            If .Rows(1).Find(What:=key, LookIn:=xlValues, LookAt:=xlWhole) Is Nothing Then
                extractWsEndCol = extractWsEndCol + 1
                .Cells(1, extractWsEndCol).value = key
                dataArr = headerKeyDict(key)
                .Cells(extractWsEndRow + 1, extractWsEndCol).Resize(UBound(dataArr), 1).value = Application.Transpose(dataArr)
            End If
        Next key


        Erase dataArr
        ReDim dataArr(1 To 1, 1 To 1)
        
        Dim headerColDict                     As Object
        
        Set headerColDict = GetHeaderAsDictionary(taxfreeExtractWs, 1)
        extractWsEndRow = .Cells(Rows.count, 1).End(xlUp).Row
            
        .Range(.Cells(2, headerColDict("과세소득_총계")), .Cells(extractWsEndRow, extractWsEndCol)).NumberFormat = "#,###"
        .Range("A1").CurrentRegion.Columns.AutoFit
        
    End With
            

    basicInfoWb.SaveAs fileName:=processingFilePath
    basicInfoWb.Close
    
    
'1-3. 매크로 파일 config시트에 다음 작업 단계 입력
    Dim nextBatchPhase          As String
    
    If batchCount = totalBatchCount Then
        nextBatchPhase = year + 1 & "_" & "1"
    Else
        nextBatchPhase = year & "_" & batchCount + 1
    End If
    configWs.Range("G6").value = nextBatchPhase
    
                
    endTime = Timer
    executionTime = endTime - startTime
    Debug.Print "실행 시간: " & executionTime & " 초"
            
End Sub
Sub taxfreePayExtract(corpName)
'[기초정보(1차)] 파일 [수당등록] 시트에 비과세수당 항목명 입력
    Dim payRegisterWs                   As Worksheet
    Dim taxfreePayDict                    As Object
    Dim targetWs                            As Worksheet
    Dim headerColDict                    As Object
    Dim endRow                              As Integer
    Dim startCol                               As Integer
    Dim endCol                                As Integer
    Dim i                                           As Integer
    
    processingFilePath = refundTargetPath & "\" & corpName & "_기초정보(1차).xlsx"
    Set basicInfoWb = Workbooks.Open(processingFilePath)
    Set payRegisterWs = basicInfoWb.Sheets("수당등록")
    Set taxfreePayDict = CreateObject("Scripting.Dictionary")
    
    endRow = payRegisterWs.Cells(Rows.count, 1).End(xlUp).Row
    If endRow > 4 Then: payRegisterWs.Range(payRegisterWs.Cells(5, 1), payRegisterWs.Cells(endRow, 4)).ClearContents           '기 입력된 데이터 삭제(재수행 고려)
    
    '[원천징수추출] 시트에서 비과세소득 세부항목명 추출
    For Each targetWs In basicInfoWb.Worksheets
        With targetWs
            If InStr(1, targetWs.Name, "원천징수추출", vbTextCompare) > 0 And _
                InStr(1, targetWs.Name, "수정", vbTextCompare) = 0 Then
                
                Set headerColDict = GetHeaderAsDictionary(targetWs, 1)
                startCol = headerColDict("세액공제 계") + 1
                endCol = .Cells(1, Columns.count).End(xlToLeft).Column
                
                For i = startCol To endCol
                    If Not taxfreePayDict.Exists(.Cells(1, i).value) Then
                        taxfreePayDict.Add .Cells(1, i).value, True
                    End If
                Next i
            End If
        End With
    Next targetWs
    
    '[수당등록] 시트에 비과세 수당명 작성
    Dim outputRow                As Integer
    Dim key                              As Variant
    
    With payRegisterWs
        outputRow = .Cells(Rows.count, 1).End(xlUp).Row + 1
        For Each key In taxfreePayDict.keys
            .Cells(outputRow, 1).value = key
            .Cells(outputRow, 2).value = "기존비과세"
            outputRow = outputRow + 1
        Next key
    End With
    
    basicInfoWb.Save
    basicInfoWb.Close False
    
    Set basicInfoWb = Nothing
    Set taxfreePayDict = Nothing
    
    
    Application.ScreenUpdating = True
    Application.DisplayStatusBar = True
    Application.DisplayAlerts = True
    
End Sub

Sub extractPAYEData(ByVal fileList As Object, ByVal startIdx As Integer, ByVal endIdx As Integer, ByRef outputDict As Object)
'개별 원천징수영수증 데이터 추출
'※ 수정사항
'  1) 주민등록번호를 .Find("주민(*")가 아닌 공백 무시 검색으로 탐색
'  2) "합계", "주(현)", "16. 계", "20. 비과세소득계"를 고정 Offset이 아닌 항목명으로 탐색
'  3) 56. 근로소득을 55. 세액감면계 + 1행으로 가정하지 않고 직접 탐색
'  4) 주요 항목 미탐색 시 어떤 파일/항목인지 오류 메시지 표시
'  5) 비과세 상세 항목명이 중복되는 경우 코드/행번호를 붙여 Dictionary 중복 오류 방지

    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    Dim targetWb                 As Workbook
    Dim targetWs                 As Worksheet
    Dim rawWbName                As String
    Dim findRng                  As Range
    Dim tempDict                 As Object
    Dim rowCount                 As Long
    Dim j                        As Long
    Dim dataArr()                As Variant
    Dim currentFile              As String

    On Error GoTo EH

    rowCount = endIdx - startIdx + 1
    If rowCount <= 0 Then GoTo SafeExit

'1-1. 연도별 원천징수영수증(엑셀) 파일에서 개인별 내역 추출
    For j = startIdx To endIdx
        rawWbName = fileList(j)
        currentFile = rawWbName

        Set targetWb = Workbooks.Open(rawWbName, ReadOnly:=True)
        Set targetWs = targetWb.Sheets(1)
        Set tempDict = CreateObject("Scripting.Dictionary")

        '개인별 원천징수영수증 파일명에서 사번 추출
        Dim frontUnderBarPos            As Long
        Dim backUnderBarPos             As Long
        Dim empNum                      As String

        If Not IsNumeric(empNumPos) Then
            empNum = ""
        Else
            backUnderBarPos = FindNth(Dir(rawWbName), "_", empNumPos)
            If empNumPos = 1 Then
                frontUnderBarPos = 0
            Else
                frontUnderBarPos = FindNth(Dir(rawWbName), "_", empNumPos - 1)
            End If

            If backUnderBarPos > frontUnderBarPos Then
                empNum = Mid$(Dir(rawWbName), frontUnderBarPos + 1, backUnderBarPos - frontUnderBarPos - 1)
            Else
                empNum = ""
            End If
        End If

        With targetWs
            Dim keySpacePos             As Long
            Dim keyResidentIDCell       As Range
            Dim residentID              As String
            Dim empName                 As String
            Dim keySumCell              As Range
            Dim keySumCol               As Long
            Dim keyPresentCell          As Range
            Dim keyPresentCol           As Long
            Dim keyTotalTaxPayCell      As Range
            Dim presentTaxPay           As Long
            Dim totalTaxPay             As Long
            Dim keyTotalTaxfreePayCell  As Range
            Dim presentTaxfreePay       As Long
            Dim totalTaxfreePay         As Long
            Dim fileIndex               As String
            Dim rightValueCell          As Range

            '============================================================
            ' 주민등록번호 / 성명
            '============================================================
            Set keyResidentIDCell = RequirePAYECell(.UsedRange, "주민등록번호", "주민등록번호", Dir(rawWbName))

            '주민등록번호가 라벨과 같은 셀에 같이 있으면 같은 셀에서 추출,
            '아니면 오른쪽 병합영역의 시작셀 값을 사용
            If InStr(1, CStr(keyResidentIDCell.value), "-", vbTextCompare) > 0 Then
                residentID = CStr(keyResidentIDCell.value)
            Else
                Set rightValueCell = NextMergedStartRight(keyResidentIDCell)
                If rightValueCell Is Nothing Then
                    Err.Raise vbObjectError + 2201, "extractPAYEData", _
                              "주민등록번호 값 셀을 찾을 수 없습니다. [" & Dir(rawWbName) & "]"
                End If
                residentID = CStr(rightValueCell.value)
            End If

            residentID = Trim$(residentID)
            keySpacePos = InStrRev(residentID, " ")
            If keySpacePos > 0 Then residentID = Mid$(residentID, keySpacePos + 1)
            residentID = Replace(residentID, " ", "")

            '성명은 주민등록번호 라벨 왼쪽의 마지막 값 셀을 우선 사용
            empName = CStr(keyResidentIDCell.End(xlToLeft).value)
            empName = Trim$(empName)
            keySpacePos = InStrRev(empName, " ")
            If keySpacePos > 0 Then empName = Mid$(empName, keySpacePos + 1)

            If Len(empName) = 0 Then
                Err.Raise vbObjectError + 2202, "extractPAYEData", _
                          "성명을 찾을 수 없습니다. [" & Dir(rawWbName) & "]"
            End If

            '============================================================
            ' 근무처별 소득명세 : 합계 / 주(현) / 16.계
            '============================================================
            Set findRng = .Range(.Cells(keyResidentIDCell.Row, 1), _
                                 .Cells(Application.Min(keyResidentIDCell.Row + 20, .Rows.count), .UsedRange.Columns(.UsedRange.Columns.count).Column))

            Set keySumCell = RequirePAYECell(findRng, "합계", "합계 컬럼", Dir(rawWbName))
            keySumCol = keySumCell.Column

            Set keyPresentCell = RequirePAYECell(findRng, "주(현)", "주(현) 컬럼", Dir(rawWbName))
            keyPresentCol = keyPresentCell.Column

            '기존: findRng.Offset(12, 0) 후 "계" 검색
            '수정: 16. 계 항목을 직접 검색
            Set keyTotalTaxPayCell = RequirePAYECell(.UsedRange, "16.계", "16. 계(과세소득 합계)", Dir(rawWbName))

            presentTaxPay = ExtractNumericOnly(.Cells(keyTotalTaxPayCell.Row, keyPresentCol).value)
            totalTaxPay = ExtractNumericOnly(.Cells(keyTotalTaxPayCell.Row, keySumCol).value)

            '============================================================
            ' 비과세소득계
            '============================================================
            Set keyTotalTaxfreePayCell = RequirePAYECell(.UsedRange, "비과세소득계", "20. 비과세소득계", Dir(rawWbName))

            presentTaxfreePay = ExtractNumericOnly(.Cells(keyTotalTaxfreePayCell.Row, keyPresentCol).value)
            totalTaxfreePay = ExtractNumericOnly(.Cells(keyTotalTaxfreePayCell.Row, keySumCol).value)

            'Solution이 '엑셀' 버전일 경우, 사번 대신 주민번호로 키 값 대체
            If Solution = "엑셀" Then
                empNum = residentID
            End If

            fileIndex = GetTextAfterLastUnderscore(Dir(rawWbName))

            AddOrSetPAYE tempDict, "기간", "20" & yearName
            AddOrSetPAYE tempDict, empNumName, empNum
            AddOrSetPAYE tempDict, empNameName, empName
            AddOrSetPAYE tempDict, resiIDName, residentID
            AddOrSetPAYE tempDict, "인덱스", fileIndex
            AddOrSetPAYE tempDict, "과세소득_총계", totalTaxPay
            AddOrSetPAYE tempDict, "비과세소득_총계", totalTaxfreePay
            AddOrSetPAYE tempDict, "과세소득_현계", presentTaxPay
            AddOrSetPAYE tempDict, "비과세소득_현계", presentTaxfreePay


            '============================================================
            ' Ⅲ세액명세 / Ⅳ정산명세
            '============================================================
            Dim firstAmountLineCol         As Long
            Dim secondAmountLineCol        As Long
            Dim earnedIncomeCell            As Range
            Dim earnedIncome                As Long
            Dim deductedIncomeCell          As Range
            Dim deductedIncome              As Long
            Dim OtherDeductIncomeSCell      As Range
            Dim OtherDeductIncomeS          As Long
            Dim taxBaseCell                 As Range
            Dim taxBase                     As Long
            Dim taxReductSumCell            As Range
            Dim taxReductSum                As Long
            Dim taxWorkDeductCell           As Range
            Dim taxWorkDeduct               As Long
            Dim taxDeductSumCell            As Range
            Dim taxDeductSum                As Long

            Set earnedIncomeCell = RequirePAYECell(.UsedRange, "근로소득금액", "23. 근로소득금액", Dir(rawWbName))
            Set rightValueCell = NextMergedStartRight(earnedIncomeCell)
            If rightValueCell Is Nothing Then
                Err.Raise vbObjectError + 2203, "extractPAYEData", _
                          "23. 근로소득금액의 금액 컬럼을 찾을 수 없습니다. [" & Dir(rawWbName) & "]"
            End If
            firstAmountLineCol = rightValueCell.Column
            earnedIncome = ExtractNumericOnly(.Cells(earnedIncomeCell.Row, firstAmountLineCol).value)

            Set deductedIncomeCell = RequirePAYECell(.UsedRange, "차감소득금액", "37. 차감소득금액", Dir(rawWbName))
            deductedIncome = ExtractNumericOnly(.Cells(deductedIncomeCell.Row, firstAmountLineCol).value)

            Set OtherDeductIncomeSCell = RequirePAYECell(.UsedRange, "그밖의소득공제계", "47. 그 밖의 소득공제 계", Dir(rawWbName))
            OtherDeductIncomeS = ExtractNumericOnly(.Cells(OtherDeductIncomeSCell.Row, firstAmountLineCol).value)

            Set taxBaseCell = RequirePAYECell(.UsedRange, "종합소득과세표준", "49. 종합소득과세표준", Dir(rawWbName))
            Set rightValueCell = NextMergedStartRight(taxBaseCell)
            If rightValueCell Is Nothing Then
                Err.Raise vbObjectError + 2204, "extractPAYEData", _
                          "49. 종합소득과세표준의 금액 컬럼을 찾을 수 없습니다. [" & Dir(rawWbName) & "]"
            End If
            secondAmountLineCol = rightValueCell.Column
            taxBase = ExtractNumericOnly(.Cells(taxBaseCell.Row, secondAmountLineCol).value)

'============================================================
' 세액감면계
'============================================================
Set taxReductSumCell = RequirePAYECell( _
    .UsedRange, _
    "세액감면계", _
    "세액감면계", _
    Dir(rawWbName) _
)

taxReductSum = ExtractNumericOnly( _
    .Cells(taxReductSumCell.Row, secondAmountLineCol).value _
)


'============================================================
' 세액공제계
'============================================================
Set taxDeductSumCell = RequirePAYECell( _
    .UsedRange, _
    "세액공제계", _
    "세액공제계", _
    Dir(rawWbName) _
)

taxDeductSum = ExtractNumericOnly( _
    .Cells(taxDeductSumCell.Row, secondAmountLineCol).value _
)


'============================================================
' 근로소득 세액공제
' 연도별 번호가 달라질 수 있으므로
' 세액감면계 ~ 세액공제계 사이에서 "근로소득" 검색
'============================================================

Dim taxWorkSearchRng As Range
Dim taxWorkCell As Range
Dim taxWorkText As String

Set taxWorkDeductCell = Nothing

Set taxWorkSearchRng = .Range( _
    .Cells(taxReductSumCell.Row + 1, 1), _
    .Cells(taxDeductSumCell.Row - 1, .UsedRange.Columns(.UsedRange.Columns.count).Column) _
)

For Each taxWorkCell In taxWorkSearchRng.Cells

    taxWorkText = CStr(taxWorkCell.value)

    '공백 / 줄바꿈 / 전각공백 제거
    taxWorkText = Replace(taxWorkText, " ", "")
    taxWorkText = Replace(taxWorkText, vbCr, "")
    taxWorkText = Replace(taxWorkText, vbLf, "")
    taxWorkText = Replace(taxWorkText, ChrW(12288), "")

    '근로소득은 찾되 "근로소득금액"은 제외
    If InStr(1, taxWorkText, "근로소득", vbTextCompare) > 0 Then

        If InStr(1, taxWorkText, "근로소득금액", vbTextCompare) = 0 Then

            Set taxWorkDeductCell = taxWorkCell
            Exit For

        End If

    End If

Next taxWorkCell


If taxWorkDeductCell Is Nothing Then

    Err.Raise _
        vbObjectError + 2205, _
        "extractPAYEData", _
        "[근로소득 세액공제] 항목을 찾을 수 없습니다. 파일: " & Dir(rawWbName)

End If


taxWorkDeduct = ExtractNumericOnly( _
    .Cells(taxWorkDeductCell.Row, secondAmountLineCol).value _
)


            '============================================================
            ' 결정세액 / 기납부세액 / 차감징수세액
            '============================================================
            Dim keySpecialTaxCell             As Range
            Dim keyIncomeTaxCell              As Range
            Dim localTaxCell                  As Range
            Dim localTaxCol                   As Long
            Dim incomTaxCol                   As Long
            Dim determinTaxRow                As Long
            Dim paidTaxRngStartRow            As Long
            Dim determinIncomTax              As Long
            Dim determinLocalTax              As Long
            Dim keyDeductTaxCell              As Range
            Dim deductIncomTax                As Long
            Dim deductLocalTax                As Long
            Dim addRowForPaidTaxRng           As Long
            Dim paidIncomTaxRng               As Range
            Dim paidLocalTaxRng               As Range
            Dim paidIncomTax                  As Long
            Dim paidLocalTax                  As Long
            Dim firstKSTAddress               As String
            Dim nextSpecialTaxCell            As Range
            Dim taxHeaderRng                  As Range

            Set keySpecialTaxCell = RequirePAYECell(.UsedRange, "농어촌특별세", "81. 농어촌특별세", Dir(rawWbName))
            firstKSTAddress = keySpecialTaxCell.Address

            Set taxHeaderRng = .Range(.Cells(keySpecialTaxCell.Row, 1), .Cells(keySpecialTaxCell.Row, .UsedRange.Columns(.UsedRange.Columns.count).Column))
            Set keyIncomeTaxCell = RequirePAYECell(taxHeaderRng, "79.소득세", "79. 소득세", Dir(rawWbName))
            Set localTaxCell = RequirePAYECell(taxHeaderRng, "80.지방소득세", "80. 지방소득세", Dir(rawWbName))

Private Function RequirePAYECell(ByVal rng As Range, _
                                 ByVal keyword As String, _
                                 ByVal itemName As String, _
                                 ByVal fileName As String) As Range
    Dim foundCell As Range

    Set foundCell = FindCellInRange(rng, keyword)

    If foundCell Is Nothing Then
        Err.Raise vbObjectError + 2301, "RequirePAYECell", _
                  "[" & itemName & "] 항목을 찾을 수 없습니다. 파일: " & fileName
    End If

    Set RequirePAYECell = foundCell
End Function

Private Sub AddOrSetPAYE(ByRef dict As Object, ByVal key As String, ByVal value As Variant)
    If dict.Exists(key) Then
        dict(key) = value
    Else
        dict.Add key, value
    End If
End Sub

Private Function NormalizePAYEText(ByVal value As Variant) As String
    Dim txt As String

    If IsError(value) Or IsNull(value) Then
        NormalizePAYEText = ""
        Exit Function
    End If

    txt = CStr(value)
    txt = Replace(txt, Chr(160), "")
    txt = Replace(txt, ChrW(12288), "")
    txt = Replace(txt, vbTab, "")
    txt = Replace(txt, vbCr, "")
    txt = Replace(txt, vbLf, "")
    txt = Replace(txt, " ", "")

    NormalizePAYEText = Trim$(txt)
End Function

                End If
            Next i

        Next r

        lastCol = .Cells(4, Columns.count).End(xlToLeft).Column
        With payAdjustWs.Range(.Cells(3, 1), .Cells(lastRow, lastCol))

            .Columns.AutoFit
        End With
        .Range(.Columns(payAdjustHeaderColDict("과세차감총액")), .Columns(lastCol)).NumberFormat = "#,###"

    basic3rdWb.Close SaveChanges:=True

Sub replaceWords(targetWs As Worksheet)
'각 문서의 치환 키워드를 사업장 정보로 대체

    Dim cell                        As Range
    Dim i                             As Integer
    
    Set targetRng = targetWs.UsedRange
    For Each cell In targetRng
        For i = LBound(replaceWordsArr, 2) To UBound(replaceWordsArr, 2)
            If Not IsEmpty(cell.value) Then
                cell.value = Replace(cell.value, replaceWordsArr(1, i), replaceWordsArr(2, i))
            End If
        Next i
    Next cell

End Sub
Function ExtractNumbers(cellValue As String) As String
'숫자만 추출
    Dim i               As Long
    Dim result      As String
    result = ""
    
    For i = 1 To Len(cellValue)
        If Mid(cellValue, i, 1) Like "[0-9]" Then
            result = result & Mid(cellValue, i, 1)
        End If
    Next i
    
    ExtractNumbers = result
End Function

Sub RenamePDFFiles(corpName As String, year As Variant)
'▶▶▶▶▶ pdf 파일명 뒤에 ("_" & 인덱스) 추가 (변경 리스트 저장 후 적용)
    Dim folderPath As String
    Dim fileName As String
    Dim fileIndex As Integer
    Dim fileSystem As Object
    Dim file As Object
    Dim newFileName As String
    Dim fileList As Object
    
    yearName = year & "년"
    
    folderPath = processPAYEpdf & "\" & yearName & "\"
    
    Set fileSystem = CreateObject("Scripting.FileSystemObject")
    
    If Not fileSystem.FolderExists(folderPath) Then
        MsgBox "지정된 폴더가 존재하지 않습니다!", vbExclamation, "오류"
        Exit Sub
    End If
    
    'PDF 파일 리스트 수집
    Set fileList = CreateObject("Scripting.Dictionary")
    
    For Each file In fileSystem.GetFolder(folderPath).files
        If LCase(fileSystem.GetExtensionName(file.Name)) = "pdf" And Not file.Name Like "~$*" Then
            fileList.Add file.Name, file.Name ' 원본 파일명 저장
        End If
    Next file
        
    '저장된 리스트 기반으로 파일명 변경
    fileIndex = 1
    
    Dim key As Variant
    For Each key In fileList.keys
        fileName = fileSystem.GetBaseName(key) ' 확장자 제외한 파일명
        newFileName = fileName & "_" & fileIndex & ".pdf"
        
        ' 파일명 변경
        Name folderPath & key As folderPath & newFileName
        
        fileIndex = fileIndex + 1
    Next key
    
    Set fileList = Nothing
    Set fileSystem = Nothing
    
End Sub
Sub RestoreOriginalPDFFilenames(corpName As String, year As Variant)
    ' pdf 파일명 뒤에 ("_" & 인덱스) 삭제 (리스트 저장 후 실행)
    Dim folderPath As String
    Dim fileName As String
    Dim fileSystem As Object
    Dim file As Object
    Dim originalFileName As String
    Dim regEx As Object
    Dim fileList As Object
    Dim match As Object
        
    Set regEx = CreateObject("VBScript.RegExp")
    regEx.Pattern = "_\d+\.pdf$" ' "_숫자.pdf" 패턴 탐색
    regEx.Global = False
    
    yearName = year & "년"
    
    folderPath = processPAYEpdf & "\" & yearName & "\"
    
    Set fileSystem = CreateObject("Scripting.FileSystemObject")
    
    ' 폴더 존재 여부 확인
    If Not fileSystem.FolderExists(folderPath) Then
        MsgBox "지정된 폴더가 존재하지 않습니다!", vbExclamation, "오류"
        Exit Sub
    End If
    
    'PDF 파일 리스트 수집
    Set fileList = CreateObject("Scripting.Dictionary")
    
    For Each file In fileSystem.GetFolder(folderPath).files
        If LCase(fileSystem.GetExtensionName(file.Name)) = "pdf" And Not file.Name Like "~$*" Then
            ' "_숫자.pdf" 형식이면 리스트에 추가
            If regEx.Test(file.Name) Then
                fileList.Add file.Name, file.Name ' 원본 파일명 저장
            End If
        End If
    Next file
    
    '저장된 리스트 기반으로 파일명 변경
    Dim key As Variant
    For Each key In fileList.keys
        ' "_숫자.pdf" 부분을 제거하여 원래 파일명 복원
        originalFileName = regEx.Replace(key, ".pdf")
        
        Name folderPath & key As folderPath & originalFileName
    Next key
    
    Set fileList = Nothing
    Set fileSystem = Nothing
    
    Set regEx = Nothing

End Sub
Sub oneRowShNameEdit(corpName)
'oneRow 시트명 "yy년MM월' 형태로 변경
    
'[급여대장(엑셀)] 폴더의 파일들 반복 작업
    Dim fso                                       As Object
    Dim folder                                  As Object
    Dim fileItem                                As Object
    Dim targetWb                            As Workbook
    Dim targetWs                             As Worksheet
    Dim targetWbNamePattern       As String
    Dim fiteredWbName                  As String
    Dim yearStr                                 As String
    Dim pos                                      As Integer
    Dim regEx                                   As Object
    Dim matches                               As Object
    Dim sheetName                           As String
    Dim extractedNumber                 As String
    
    targetWbNamePattern = "*oneRow*"
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    Set folder = fso.GetFolder(processPayrollPath)
    
    
    For Each fileItem In folder.files
        If LCase(fileItem.Name) Like LCase(targetWbNamePattern) And fileItem.Name Like "*.xls*" Then
            '파일명에서 연도 정보 추출 'yy년'
            pos = InStr(fileItem.Name, "년")

            If pos > 2 Then
                yearStr = Mid(fileItem.Name, pos - 2, 3)
            Else
                yearStr = ""
            End If
            
            Set targetWb = Workbooks.Open(fileItem.Path)
            '시트명에서 월 정보 추출 'MM월'
            For Each targetWs In targetWb.Worksheets

                sheetName = targetWs.Name
                
                Set regEx = CreateObject("VBScript.RegExp")
                ' 패턴 설명:
                ' (?:^|[^0-9])   : 숫자가 아닌 문자 또는 문자열 시작이어야 함
                ' \s*            : 공백이 있을 수 있음
                ' (\d{1,2})      : 1자리 또는 2자리 숫자를 캡처
                ' 월             : 바로 "월"이 와야 함
                regEx.Pattern = "(?:^|[^0-9])(\d{1,2})월"
                regEx.IgnoreCase = False
                regEx.Global = False
                
                If regEx.Test(sheetName) Then
                    Set matches = regEx.Execute(sheetName)
                    extractedNumber = matches(0).SubMatches(0)
                    
                    ' 1자리인 경우 왼쪽에 0 추가
                    If Len(extractedNumber) = 1 Then
                        extractedNumber = "0" & extractedNumber
                    End If
                Else
                    extractedNumber = ""
                End If
                '시트명 변경
                targetWs.Name = yearStr & extractedNumber & "월"
                
            Next targetWs
            
            targetWb.Close SaveChanges:=True
            
        End If
    Next fileItem

End Sub

    Do While fileName <> ""

        With sourceWb.Sheets(1)

            Set dataRng = .UsedRange
            On Error GoTo 0
            
            If Not dataRng Is Nothing Then
                ' 첫 번째 파일이면 헤더 포함 복사
                If headerCopied = False Then
                    dataRng.Copy masterWs.Cells(masterRow, 1)

            End If

Sub extractIncomeTaxForXlsVer(corpName As String, year As Variant)
'1.대상자 기존 원천징수영수증 데이터 추출/정리(for엑셀버전)
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.DisplayStatusBar = False

    Dim basic3rdWb                         As Workbook
    Dim payAdjustWs                       As Worksheet
    Dim payAdjustHeaderColDict     As Object

    Set basic3rdWb = Workbooks.Open(inputPath & "\" & corpName & "_기초정보(3차).xlsx")
    Set payAdjustWs = basic3rdWb.Sheets("수당조정(비과세)")
    Set payAdjustHeaderColDict = GetHeaderAsDictionary(payAdjustWs, 4)
    
    
'1-1.[수당조정(비과세)]시트에서 환급대상자 개인별 비과세급여 정보 추출
    Dim payAdjustCol                     As Integer
    Dim targetArr()                         As Variant
    Dim targetCount                      As Integer
    Dim empKey                              As String
    
    With payAdjustWs
        payAdjustCol = 8
        endRow = .Cells(Rows.count, 1).End(xlUp).Row
        endCol = .Cells(4, Columns.count).End(xlToLeft).Column
        prevTaxfreePayCountX2 = endCol - payAdjustHeaderColDict("과세차감총액") - 1   '기존 비과세급여 수 x 2(총계,현계)

        If endCol > payAdjustHeaderColDict("과세차감총액") Then
            targetArr = .Range("A5:A" & endRow).value
            For r = 1 To UBound(targetArr, 1)
                If targetArr(r, 1) = "20" & yearName Then targetCount = targetCount + 1
            Next r
                
            ReDim personalArr(1 To targetCount, 1 To payAdjustCol - 2)
            ReDim adjustTfPayArr(1 To targetCount, 1 To 6)
            ReDim prevTaxfreeArr(1 To targetCount, 1 To prevTaxfreePayCountX2 * 3)
            
            targetCount = 0
            For r = 5 To endRow
                Dim taxFreeName                      As String
                Dim taxFreeOriName                   As String
                Dim taxFreeValue                       As Double
                Dim p          As Integer, s            As Integer
                Dim taxReductRate                    As Double
            
                If .Cells(r, 1).value = "20" & yearName Then
                    targetCount = targetCount + 1
                    '인적사항 추출
                    For i = 1 To payAdjustCol - 2
                        personalArr(targetCount, i) = .Cells(r, i).value
                    Next i
                    '조정비과세(야간연장수당) 추출
                    adjustTfPayArr(targetCount, 1) = .Cells(3, payAdjustCol).value      '조정비과세 명칭
                    adjustTfPayArr(targetCount, 2) = .Cells(2, payAdjustCol).value      '조정비과세 코드
                    adjustTfPayArr(targetCount, 3) = .Cells(r, payAdjustCol).value       '조정비과세 값
                    '기존비과세 추출
                    For i = 1 To prevTaxfreePayCountX2
                        If .Cells(r, 8 + i).value <> "" Then
                            taxFreeName = .Cells(3, payAdjustCol + i).value                                               '비과세/감면 명칭
                            taxFreeOriName = .Cells(4, payAdjustCol + i).value                                            '비과세/감면 원천 명칭
                            taxFreeValue = .Cells(r, payAdjustCol + i).value                                              '비과세/감면 값

                            prevTaxfreeArr(targetCount, (3 * i - 1)) = .Cells(2, payAdjustCol + i).value    '비과세/감면 코드
                            prevTaxfreeArr(targetCount, (3 * i)) = taxFreeValue
                            
                            '조특법 감면 금액이 있을 경우 "%" 바로 앞 숫자만 추출
                            If InStr(taxFreeName, "감면") > 0 And InStr(taxFreeName, "%") > 0 And taxFreeValue > 0 Then
                                p = InStr(taxFreeName, "%")
                                s = p - 1
                                Do While s >= 1 And Mid$(taxFreeName, s, 1) Like "[0-9]"
                                    s = s - 1

Function InsertImagesFromFolder(imgFolder As String, ws As Worksheet) As Integer
'폴더에서 이미지 파일을 순서대로 시트에 입력
'엑셀 2019 기준: 크롭/왜곡 없이 원본 비율 유지 + 박스 안에 완전히 들어오게 맞춤
'세로(Top)는 상단 정렬, 가로는 중앙 정렬

    Dim fso As Object, folder As Object, file As Object
    Dim imgDict As Object, imgNum As Integer
    Dim colIndex As Integer, key As Variant
    Dim targetCell As Range, mergedArea As Range
    Dim imgCount As Integer
    Dim shp As Shape
    Dim shapeName As String

    Dim widthCM As Double, heightCM As Double
    Dim boxW As Double, boxH As Double
    Dim origW As Double, origH As Double
    Dim scaleF As Double

    ' 1) 삽입한 기존 이미지 삭제 (이름 패턴: "InsertedImage_")
    For Each shp In ws.Shapes
        If shp.Type = msoPicture Then
            If Left(shp.Name, 14) = "InsertedImage_" Then
                shp.Delete
            End If
        End If
    Next shp

    ' 2) 파일 시스템 객체 준비
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FolderExists(imgFolder) Then
        InsertImagesFromFolder = 0
        Exit Function
    End If

    ' 3) 이미지 파일 수집
    Set imgDict = CreateObject("Scripting.Dictionary")
    imgCount = 0
    Set folder = fso.GetFolder(imgFolder)

    For Each file In folder.files
        imgNum = CInt(GetTextAfterLastUnderscore(file.Name))
        If imgNum > 0 Then
            imgDict(imgNum) = file.Path
            imgCount = imgCount + 1
        End If
    Next file

    ' 4) 목표 박스 크기 (cm → point)  ★엑셀 내장 변환 사용
    widthCM = 49
    heightCM = 70
    boxW = Application.CentimetersToPoints(widthCM)
    boxH = Application.CentimetersToPoints(heightCM)

    ' 5) 이미지 삽입
    colIndex = 1

    For Each key In imgDict.keys
        Set targetCell = ws.Cells(1, colIndex)
        Set mergedArea = targetCell.MergeArea

        shapeName = "InsertedImage_" & colIndex

        ' ? 원본 크기로 삽입
        Set shp = ws.Shapes.AddPicture( _
            fileName:=imgDict(key), _
            LinkToFile:=msoFalse, _
            SaveWithDocument:=msoTrue, _
            Left:=mergedArea.Left, _
            Top:=mergedArea.Top, _
            Width:=-1, _
            Height:=-1)

        With shp
            .Name = shapeName
            .Placement = xlMoveAndSize
            .ZOrder msoSendToBack

            ' 원본 크기 저장
            origW = .Width
            origH = .Height

            ' ? 박스 안에 "완전히 들어가게" 스케일 팩터 (MIN)
            scaleF = WorksheetFunction.Min(boxW / origW, boxH / origH)

            ' ? 같은 팩터로 가로/세로 동시에 스케일 (미세오차 최소화)
            .ScaleWidth Factor:=scaleF, RelativeToOriginalSize:=msoTrue
            .ScaleHeight Factor:=scaleF, RelativeToOriginalSize:=msoTrue

            ' (선택) 렌더링 타이밍 잔오차 방지용으로 한 번 더 스냅
            If Abs(.Width - (origW * scaleF)) > 0.05 Or Abs(.Height - (origH * scaleF)) > 0.05 Then
                .ScaleWidth Factor:=scaleF, RelativeToOriginalSize:=msoTrue
                .ScaleHeight Factor:=scaleF, RelativeToOriginalSize:=msoTrue
            End If

            ' ? 가로는 중앙 정렬
            .Left = mergedArea.Left + (boxW - .Width) / 2

            ' ? 세로는 상단 정렬 (Top align)
            .Top = mergedArea.Top
        End With

        colIndex = colIndex + 1
    Next key

    InsertImagesFromFolder = imgCount
End Function

Sub WaitForPDFs(folderPath As String, expectedCount As Integer, Optional timeout As Double = 10)
    Dim startTime As Double
    Dim fso As Object, folder As Object, file As Object
    Dim fileCount As Integer
    
    ' FileSystemObject 생성
    Set fso = CreateObject("Scripting.FileSystemObject")
    startTime = Timer ' 현재 시간 저장

    ' 폴더가 존재하는지 확인
    If Not fso.FolderExists(folderPath) Then Exit Sub

    ' PDF가 지정된 개수만큼 저장될 때까지 대기 (최대 timeout 초)
    Do
        ' 현재 폴더 내 PDF 개수 확인
        fileCount = 0
        Set folder = fso.GetFolder(folderPath)
        For Each file In folder.files
            If LCase(Right(file.Name, 4)) = ".pdf" Then fileCount = fileCount + 1
        Next file

        ' 기대한 개수만큼 생성되면 종료
        If fileCount >= expectedCount Then Exit Do

        ' 타임아웃 체크
        If Timer - startTime >= timeout Then Exit Do

        DoEvents ' UI 멈춤 방지
    Loop
End Sub

Sub filterSubmitPAYE(corpName As String, year As Variant, Solution As String)
'2. [원천징수(PDF)] 수정전/후 폴더를 복사 후 제출용 폴더에 이동 후 대상자 파일만 남김
    
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.DisplayStatusBar = False
    
    Debug.Print "filterSubmitPAYE : " & yearName

    Dim fso                                 As Object
    Dim file                                 As Object
    Dim fileName                        As String
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    
'2-1. 원천징수영수증 폴더 복사하여 제출용 폴더에 이동
    ' 수정전 폴더 복사/이동
    If fso.FolderExists(processPAYEpdf & "\" & yearName) Then
        If fso.FolderExists(asisPAYEpath) Then fso.DeleteFolder asisPAYEpath, True
        fso.CopyFolder processPAYEpdf & "\" & yearName, outputPAYEtarget & "\"
        fso.MoveFolder outputPAYEtarget & "\" & yearName, asisPAYEpath
    End If

    ' 수정후 폴더 복사/이동
    If fso.FolderExists(outputPAYEpdf & "\" & yearName) And Solution <> "엑셀" Then
        If fso.FolderExists(tobePAYEpath) Then fso.DeleteFolder tobePAYEpath, True
        fso.CopyFolder outputPAYEpdf & "\" & yearName, outputPAYEtarget & "\"
        fso.MoveFolder outputPAYEtarget & "\" & yearName, tobePAYEpath
    End If


    Dim folderPaths      As Variant
    Dim folderPath       As Variant
    Dim folder              As Object

    Dim fileToCheck     As String
    Dim validIndex       As Long
    Dim validName       As String
    Dim newFileName  As String
    Dim suffix               As String
    Dim isValid              As Boolean
    Dim i                        As Integer

    Dim files()              As Object
    Dim fileCount        As Long
    Dim fileIndex         As Long

'2-2. 환급대상 파일만 남기고 삭제
    If Solution <> "엑셀" Then
        folderPaths = Array(asisPAYEpath, tobePAYEpath)
    Else
        folderPaths = Array(asisPAYEpath)       '"엑셀" 버전일 경우 "수정전" 폴더만 작업
    End If

    '환급대상 아닌 파일 삭제 (역순 반복)
    For Each folderPath In folderPaths
        If fso.FolderExists(folderPath) Then
            Set folder = fso.GetFolder(folderPath)

            ' 파일 객체를 배열에 저장
            fileCount = folder.files.count
            ReDim files(1 To fileCount)
            fileIndex = 1
            For Each file In folder.files
                Set files(fileIndex) = file
                fileIndex = fileIndex + 1
            Next file

            ' 역순으로 배열 순회하며 파일 삭제
            For fileIndex = fileCount To 1 Step -1
                Set file = files(fileIndex)
                fileName = file.Name
                fileToCheck = ""

                ' 파일명에서 사번 추출(엑셀 버전일 경우 "인덱스" 추출)
                If Solution = "더존" Then
                    If InStr(fileName, "_") > 0 Then
                        fileToCheck = Left(fileName, InStr(fileName, "_") - 1)
                    End If
                ElseIf Solution = "세무사랑" Then
                    If InStr(fileName, "_") > 0 Then
                        fileToCheck = Split(fileName, "_")(1)
                    End If
                ElseIf Solution = "엑셀" Then
                    If InStr(fileName, "_") > 0 Then
                        fileToCheck = Mid(fileName, InStrRev(fileName, "_") + 1)
                    End If
                End If

                '비대상자 파일 삭제
                If fileToCheck <> "" Then
                    isValid = False
                    If Solution = "엑셀" Then
                        For i = LBound(targetIndexArr, 2) To UBound(targetIndexArr, 2)
                            If targetIndexArr(1, i) & ".pdf" = fileToCheck Then        '인덱스
                                isValid = True
                                Exit For
                            End If
                        Next i
                    Else
                        For i = LBound(correctArr1, 2) To UBound(correctArr1, 2)
                            If correctArr1(1, i) = fileToCheck Then         '사번
                                isValid = True
                                Exit For
                            End If
                        Next i
                    End If

                    If Not isValid Then
                        file.Delete
                    End If
                Else
                    file.Delete ' 사번 추출 실패한 파일도 삭제
                End If
            Next fileIndex
        Else
            MsgBox "폴더가 존재하지 않습니다: " & folderPath, vbExclamation
        End If
    Next folderPath


'2-3. 대상자 파일명 변경
    For Each folderPath In folderPaths
        If fso.FolderExists(folderPath) Then
            Set folder = fso.GetFolder(folderPath)

            ' 파일 객체를 배열에 저장
            fileCount = folder.files.count
            ReDim files(1 To fileCount)
            fileIndex = 1
            For Each file In folder.files
                Set files(fileIndex) = file
                fileIndex = fileIndex + 1
            Next file

            ' 배열 순회하며 파일명 변경
            For fileIndex = 1 To fileCount
                Set file = files(fileIndex)
                fileName = file.Name
                fileToCheck = ""
                validIndex = -1
                validName = ""
                
                ' 파일명에서 사번 추출(엑셀 버전일 경우 "인덱스" 추출)
                If Solution = "더존" Then
                    If InStr(fileName, "_") > 0 Then
                        fileToCheck = Left(fileName, InStr(fileName, "_") - 1)
                    End If
                ElseIf Solution = "세무사랑" Then
                    If InStr(fileName, "_") > 0 Then
                        fileToCheck = Split(fileName, "_")(1)
                    End If
                ElseIf Solution = "엑셀" Then
                    If InStr(fileName, "_") > 0 Then
                        fileToCheck = Mid(fileName, InStrRev(fileName, "_") + 1)
                    End If
                End If

                If fileToCheck <> "" Then
                    If Solution = "엑셀" Then
                        For i = LBound(targetIndexArr, 2) To UBound(targetIndexArr, 2)
                            If targetIndexArr(1, i) & ".pdf" = fileToCheck Then       '인덱스
                                validIndex = targetIndexArr(2, i)   '연번
                                validName = correctArr1(2, i)
                                Exit For
                            End If
                        Next i
                    Else
                        For i = LBound(correctArr1, 2) To UBound(correctArr1, 2)
                            If correctArr1(1, i) = fileToCheck Then    '사번
                                validIndex = targetIndexArr(2, i)   '연번
                                validName = correctArr1(2, i)
                                Exit For
                            End If
                        Next i
                    End If

                    If validIndex > -1 Then
                        suffix = IIf(folderPath = asisPAYEpath, "수정전", "수정후")
                        newFileName = folder.Path & "\" & validIndex & "_" & validName & "_20" & yearName & "_원천징수영수증(" & suffix & ").pdf"
                        file.Name = Mid(newFileName, InStrRev(newFileName, "\") + 1)
                    End If
                End If
            Next fileIndex
        Else
            MsgBox "폴더가 존재하지 않습니다: " & folderPath, vbExclamation
        End If
    Next folderPath
    
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.DisplayStatusBar = True
    
End Sub

    Dim searchTitleRng      As Range
    Dim yearMonthSource     As String
    Dim yearPos             As Integer
    Dim monthPos            As Integer
    Dim title               As String

'1-1. [oneRow급여대장] 폴더 생성
    If Dir(refundTargetPath, vbDirectory) = "" Then MkDir (refundTargetPath)
    If Dir(processPayrollPath, vbDirectory) = "" Then MkDir (processPayrollPath)

    Call DeleteFilesInFolder(processPayrollPath)    '해당 폴더에 있는 파일 모두 삭제

'1-2. raw 급여대장 → oneRow급여대장 파일로 변환
    targetWbNamePattern = "*.xls*"

    Dim fileList As Collection
    Set fileList = GetFileList(rawPayrollPath, targetWbNamePattern) '★ Dir 대신 선 목록화

    Dim f As Variant
    For Each f In fileList
        rawWbName = CStr(f)
        Set targetWb = Workbooks.Open(rawPayrollPath & "\" & rawWbName)

        For Each targetWs In targetWb.Sheets
            ' -----------------------------
            ' (기존 로직) 타이틀 셀 탐색
            ' -----------------------------
            With targetWs
                .UsedRange.UnMerge
                Set searchTitleRng = .Range(.Cells(1, 1), .Cells(5, 26))
                Set titleCell = Nothing

                ' 1) "급여대장" 포함 셀 탐색
                For Each cell In searchTitleRng
                    If CStr(cell.value) Like "*급*여*대장*" Then
                        Set titleCell = cell
                        yearMonthSource = CStr(cell.value)
                        Exit For
                    End If
                Next cell

                ' 2) "귀속" 포함 셀 탐색
                If titleCell Is Nothing Then
                    For Each cell In searchTitleRng
                        If CStr(cell.value) Like "*귀속*" Then
                            Set titleCell = cell
                            yearMonthSource = CStr(cell.value)
                            Exit For
                        End If
                    Next cell
                End If

                ' 3) "인적사항" 포함 셀 탐색 (없으면 파일명에서)
                If titleCell Is Nothing Then
                    For Each cell In searchTitleRng
                        If CStr(cell.value) Like "*인적*사항*" Then
                            Set titleCell = cell
                            yearMonthSource = rawWbName   '이 경우는 년월 정보가 없어 파일명에서 가져와야 함
                            Exit For
                        End If
                    Next cell
                End If

                If titleCell Is Nothing Then
                    ' 최종 실패 시 에러 보고 후 다음 시트로
                    MsgBox "에러 : titleCell과 년월 정보를 찾지 못했습니다. 파일: " & rawWbName & ", 시트: " & .Name, vbCritical
                    GoTo NextSheet
                End If

                ' -----------------------------
                ' (기존 로직) 타이틀(년월) 파싱
                ' -----------------------------
                yearMonthSource = Replace(yearMonthSource, " ", "")
                yearPos = InStr(yearMonthSource, "년") - 2
                monthPos = InStr(yearMonthSource, "월")
                If yearPos < 1 Or monthPos < 1 Or monthPos <= yearPos Then
                    ' 파싱 실패 시 파일명에서 재시도 (간단 처리)
                    yearMonthSource = Replace(rawWbName, " ", "")
                    yearPos = InStr(yearMonthSource, "년") - 2
                    monthPos = InStr(yearMonthSource, "월")
                End If

                If yearPos >= 1 And monthPos > yearPos Then
                    title = Mid(yearMonthSource, yearPos, monthPos - yearPos + 1)
                Else
                    ' 그래도 실패 시 시트명 그대로 사용
                    title = .Name
                End If

                If Len(title) = 5 Then
                    title = Left(title, 3) & "0" & Right(title, 2)
                End If

                On Error Resume Next
                .Name = title
                On Error GoTo ErrHandler
            End With

            ' =========================
            ' ★ 추가 1: 하드코딩 컬럼 삭제
            '   - 필요 열 번호를 변경하세요.
            ' =========================
            Dim colsToDelete As Variant
            colsToDelete = Array(6, 8, 9, 10, 12, 17, 18, 20, 23, 24, 26) ' ◀◀◀ 여기만 수정해서 사용 (예: Array(3,6,7,15))
            If IsArray(colsToDelete) Then
                DeleteColumnsByNumbers targetWs, colsToDelete
            End If

            ' =========================
            ' ★ 추가 2: title(시트명)로 개별 파일 저장
            '   - 저장 경로: processPayrollPath
            '   - 파일명 충돌 시 _01, _02 ... 자동 부여
            '   - 해당 시트만 새로운 통합문서로 복사하여 .xlsx로 저장
            ' =========================
            Dim outFile As String
            outFile = processPayrollPath & "\" & SanitizeFileName(targetWs.Name & "_급상여대장") & ".xlsx"
            outFile = EnsureUniquePath(outFile)

            targetWs.Copy   ' 해당 시트를 새 통합문서로 복사
            With ActiveWorkbook
                .SaveAs fileName:=outFile, FileFormat:=xlOpenXMLWorkbook ' .xlsx
                .Close SaveChanges:=False
            End With

NextSheet:
        Next targetWs

        targetWb.Close SaveChanges:=False
    Next f

CleanUp:
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.DisplayStatusBar = True
    Exit Sub

ErrHandler:
    ' 오류 시에도 화면/알림 상태 복구
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.DisplayStatusBar = True
    MsgBox "오류가 발생했습니다: " & Err.Number & " - " & Err.Description, vbCritical
End Sub

'========================================
' 헬퍼: Dir를 한 번만 사용해 파일 목록 수집
'========================================
Private Function GetFileList(folderPath As String, filePattern As String) As Collection
    Dim col As New Collection
    Dim fName As String
    fName = Dir(folderPath & "\" & filePattern)
    Do While fName <> ""
        col.Add fName
        fName = Dir
    Loop
    Set GetFileList = col
End Function

'------------------------------------------
' 헬퍼: 파일명에 사용할 수 없는 문자 치환
'------------------------------------------
Private Function SanitizeFileName(ByVal s As String) As String
    Dim badChars As Variant, ch As Variant
    badChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For Each ch In badChars
        s = Replace(s, ch, "_")
    Next ch
    s = Trim(s)
    If s = "" Then s = "untitled"
    SanitizeFileName = s
End Function

'------------------------------------------
' 헬퍼: 동일 파일명 존재 시 _01, _02 ...로 유니크 경로 반환
'------------------------------------------
Private Function EnsureUniquePath(ByVal fullPath As String) As String
    If Dir(fullPath) = "" Then
        EnsureUniquePath = fullPath
        Exit Function
    End If

    Dim dotPos As Long, basePath As String, ext As String, i As Long
    dotPos = InStrRev(fullPath, ".")
    If dotPos > 0 Then
        basePath = Left(fullPath, dotPos - 1)
        ext = Mid(fullPath, dotPos)
    Else
        basePath = fullPath
        ext = ""
    End If

    i = 1
    Do
        EnsureUniquePath = basePath & "_" & Format(i, "00") & ext
        i = i + 1
    Loop While Dir(EnsureUniquePath) <> ""
End Function

'------------------------------------------
' 헬퍼: 지정한 열 번호들을 역순으로 삭제 (견고 버전)
'  - 1행이 비어있어도 작동하도록 마지막 열을 Find로 계산
'  - 자동필터가 켜져 있으면 해제
'  - 중복/범위 정리 후 내림차순으로 안전 삭제
'------------------------------------------
Private Sub DeleteColumnsByNumbers(ByVal ws As Worksheet, ByVal cols As Variant)
    On Error GoTo SafeExit

    ' 0) 자동필터 해제 (필터 상태에서 열 삭제시 예외 방지)
    If ws.AutoFilterMode Then
        On Error Resume Next
        ws.ShowAllData              ' 필터가 없는데 호출하면 에러 → 무시
        On Error GoTo 0
        ws.AutoFilterMode = False
    End If

    ' 1) 실제 마지막 열 계산 (시트 전체에서 마지막 사용 셀 기준)
    Dim lastUsed As Range
    Dim maxCol As Long
    Set lastUsed = ws.Cells.Find(What:="*", LookIn:=xlFormulas, _
                                 SearchOrder:=xlByColumns, SearchDirection:=xlPrevious)
    If lastUsed Is Nothing Then
        Exit Sub  ' 시트가 비어있음
    Else
        maxCol = lastUsed.Column
    End If

    ' 2) 유효한 컬럼만 모아서 내림차순 정렬 (중복 제거)
    Dim tmp As Object: Set tmp = CreateObject("Scripting.Dictionary")
    Dim i As Long, c As Long
    For i = LBound(cols) To UBound(cols)
        If IsNumeric(cols(i)) Then
            c = CLng(cols(i))
            If c >= 1 And c <= maxCol Then
                tmp(CStr(c)) = c
            End If
        End If
    Next i
    If tmp.count = 0 Then Exit Sub

    Dim keys() As Variant
    keys = tmp.Items
    ' 내림차순 정렬
    Dim j As Long, k As Long, t As Variant
    For j = LBound(keys) To UBound(keys) - 1
        For k = j + 1 To UBound(keys)
            If CLng(keys(j)) < CLng(keys(k)) Then
                t = keys(j): keys(j) = keys(k): keys(k) = t
            End If
        Next k
    Next j

    ' 3) 큰 번호부터 삭제
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    On Error Resume Next
    For i = LBound(keys) To UBound(keys)
        ws.Columns(CLng(keys(i))).EntireColumn.Delete
    Next i
    On Error GoTo 0

SafeExit:
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
End Sub
